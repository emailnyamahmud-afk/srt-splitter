    78→def _media_process_error(tool: str, returncode: int, stderr: bytes, *, paths=()) -> str:
    79→    """Keep the actionable end of native diagnostics without leaking paths."""
    80→    from core.scrub import scrub_text
    81→    detail = stderr.decode(errors="replace")
    82→    # Input/output may live outside home directories (mounted media, Windows
    83→    # drive roots). Remove the exact command paths before generic scrubbing.
    84→    for path in sorted((str(p) for p in paths if p), key=len, reverse=True):
    85→        for variant in {path, path.replace("\\", "/"), path.replace("/", "\\")}:
    86→            detail = detail.replace(variant, "[redacted path]")
    87→    detail = scrub_text(detail).strip()
    88→    tail = detail[-2000:]
    89→    if len(detail) > 2000:
    90→        tail = "…" + tail
    91→    return f"{tool} exited with code {returncode}" + (f": {tail}" if tail else ". No diagnostic output.")

   107→def _discard_invalid_source_copy(job_dir: str, media_path: str) -> None:
   108→    """Free a failed ingest copy, never a source outside an app-owned job dir."""
   109→    job = os.path.realpath(job_dir)
   110→    media = os.path.realpath(media_path)
   111→    if (
   112→        os.path.normcase(os.path.dirname(job)) != os.path.normcase(os.path.realpath(DUB_DIR))
   113→        or os.path.normcase(os.path.dirname(media)) != os.path.normcase(job)
   114→        or not os.path.basename(media).startswith("original.")
   115→    ):
   116→        return

   123→# ── Module-level state ──────────────────────────────────────────────────────
   124→# These used to live in dub_core.py. The router now re-exports them for
   125→# backward compat during the transition.
   126→
   127→_dub_jobs: dict[str, dict] = {}
   128→# Re-entrant: `save_job` takes this lock itself (see below), and the atomic
   129→# helpers call it while already holding it.
   130→_dub_jobs_lock = threading.RLock()
   131→
   132→#: Ingests currently running. Used only so "clear history" can also sweep a job
   133→#: that has no row yet — it would appear in no id list otherwise.
   134→_inflight_jobs: set[str] = set()
   135→
   136→#: Recently-deleted job ids, most-recent last. Guarded by ``_dub_jobs_lock``.
   137→#:
   138→#: Dict membership alone cannot express "the user withdrew this job" (#1252
   139→#: review): a job's FIRST persistence creates the entry, so an absent key means
   140→#: "not written yet" for a new job and "deleted" for an established one — two
   141→#: opposite instructions from one signal.
   142→#:
   143→#: Scoped to DELETED, not to in-flight ingests. Scoping it to ingests looked
   144→#: right and closed nothing that mattered: a dub is imported once and rendered
   145→#: many times, so the realistic delete lands during a RENDER, long after its
   146→#: ingest ended — and a render's save would then write the row straight back.
   147→#:
   148→#: Retained rather than cleared on completion, because there is no moment at
   149→#: which a delete stops mattering: any operation still holding that job can
   150→#: persist it. ``begin_ingest`` drops an id explicitly, since re-importing is a
   151→#: deliberate revival.
   152→#:
   153→#: Expired by AGE, not by count. A count-bounded LRU is evictable by ordinary
   154→#: use: ``DELETE /dub/history`` purges every row with no limit, so a user
   155→#: clearing a large history mid-render would push the rendering job's own
   156→#: marker out and the render would then write it back (#1252 review). Age
   157→#: cannot be gamed that way — what matters is how long ago the delete happened,
   158→#: not how many others followed it.
   159→#:
   160→#: The count cap is a memory backstop only, set far above any real history:
   161→#: ~4096 short ids is a few hundred KB. Reaching it needs 4096 deletions inside
   162→#: one TTL window, at which point the oldest markers are the least likely to
   163→#: still be held.
   164→_WITHDRAWN_TTL_S = 6 * 3600   # outlives any realistic render or transcribe
   165→_WITHDRAWN_MAX = 4096         # memory backstop, not the eviction policy
   166→_withdrawn_jobs: "OrderedDict[str, float]" = OrderedDict()
   167→
   168→_DUB_DIR_REAL = os.path.realpath(DUB_DIR)
   169→_HASH_BUF_SIZE = 1 << 18  # 256 KB chunks for hashing
   170→

   175→def compute_file_hash(path: str) -> str:
   176→    """SHA-256 digest of a file, streamed in 256 KB chunks."""
   177→    h = hashlib.sha256()
   178→    with open(path, "rb") as f:
   179→        while True:
   180→            chunk = f.read(_HASH_BUF_SIZE)
   181→            if not chunk:
   182→                break
   183→            h.update(chunk)
   184→    return h.hexdigest()

   187→def safe_job_dir(job_id: str) -> Optional[str]:
   188→    """Resolve a job directory under DUB_DIR, rejecting traversal."""
   189→    if not job_id or "/" in job_id or "\\" in job_id or job_id in (".", ".."):
   190→        return None
   191→    candidate = os.path.realpath(os.path.join(DUB_DIR, job_id))
   192→    if not candidate.startswith(_DUB_DIR_REAL + os.sep):
   193→        return None
   194→    return candidate

   197→def job_dir_referenced_by_others(job_id: str) -> "list[str]":
   198→    """History ids of OTHER jobs whose persisted paths point into ``job_id``'s
   199→    directory (#1331, the deletion half).
   200→
   201→    The content-hash cache legitimately points a newer job's ``vocals_path``
   202→    (and, for jobs created before the job-scoped-clones fix, its clone
   203→    reference paths) into an older job's directory. Deleting that older entry
   204→    used to ``rmtree`` the dir regardless, silently breaking the newer job:
   205→    single-segment regens fell back to the default voice, stems exports lost
   206→    their sources. The caller uses this to keep the DIRECTORY while still
   207→    deleting the history row — disk is the cheap thing here; another job's
   208→    voice is not.
   209→
   210→    Scans persisted ``job_data`` as text for the dir prefix rather than
   211→    enumerating every path-bearing key: keys have grown before (vocals,
   212→    no_vocals, thumb, clone refs, segment refs) and a scan cannot fall behind
   213→    the schema.
   214→    """
   215→    target = safe_job_dir(job_id)
   216→    if not target:
   217→        return []
   218→    needle = target.rstrip(os.sep) + os.sep
   219→    # JSON-encoded job_data escapes backslashes, so match the Windows form too.
   220→    needle_json = needle.replace("\\", "\\\\")
   221→    holders: list[str] = []
   222→    try:
   223→        with db_conn() as conn:
   224→            rows = conn.execute(
   225→                "SELECT id, job_data FROM dub_history WHERE id != ?", (job_id,)
   226→            ).fetchall()
   227→    except Exception:
   228→        return []  # no DB, nothing persisted can reference us
   229→    for row in rows:
   230→        data = row["job_data"] or ""
   231→        if needle in data or needle_json in data:
   232→            holders.append(row["id"])
   233→    return holders

   249→def find_cached_job(content_hash: str, exclude_job_id: str) -> Optional[dict]:
   250→    """Find a previous job with the same content hash that has the heavy
   251→    artifacts (vocals/no-vocals/scene cuts) still on disk. Returns a dict
   252→    of paths + metadata the caller can shallow-copy into the new job, or
   253→    None if no usable cache exists.
   254→    """
   255→    if not content_hash:
   256→        return None
   257→    with db_conn() as conn:
   258→        rows = conn.execute(
   259→            "SELECT id, job_data FROM dub_history WHERE content_hash=? AND id!=? ORDER BY created_at DESC LIMIT 5",
   260→            (content_hash, exclude_job_id),
   261→        ).fetchall()
   262→    for row in rows:
   263→        try:
   264→            job = json.loads(row["job_data"])
   265→        except (json.JSONDecodeError, TypeError):
   266→            continue
   267→        cached_dir = safe_job_dir(row["id"])
   268→        if not cached_dir or not os.path.isdir(cached_dir):
   269→            continue
   270→        vocals = job.get("vocals_path") or os.path.join(cached_dir, "vocals.wav")
   271→        if not os.path.isfile(vocals):
   272→            continue
   273→        # Separation-quality gate: stems produced before the HQ-extraction
   274→        # change were separated from the 16 kHz MONO ASR file — a mono,
   275→        # 8 kHz-ceiling music bed. audio_hq.wav in the cached job dir is the
   276→        # marker that its stems came from the full-quality stereo extraction;
   277→        # without it, reusing the cache would silently keep serving the
   278→        # narrow-band mono bed forever for that video. Re-separating once is
   279→        # the better deal.
   280→        if not os.path.isfile(os.path.join(cached_dir, "audio_hq.wav")):
   281→            logger.info(
   282→                "cache candidate %s has pre-HQ (mono/16k-derived) stems — "
   283→                "skipping reuse so separation reruns at full quality", row["id"],
   284→            )
   285→            continue
   286→        return {
   287→            "job_dir": cached_dir,
   288→            "job_id": row["id"],
   289→            "vocals_path": vocals,
   290→            "no_vocals_path": job.get("no_vocals_path"),
   291→            "scene_cuts": job.get("scene_cuts") or [],
   292→            "thumb_path": job.get("thumb_path"),
   293→            "duration": job.get("duration", 0.0),
   294→        }
   295→    return None

   306→def get_job(job_id: str) -> Optional[dict]:
   307→    """Look up a job. Checks the in-memory cache first, then falls back to
   308→    `dub_history.job_data` so saved projects still resolve after restart.
   309→    """
   310→    with _dub_jobs_lock:
   311→        if job_id in _dub_jobs:
   312→            return _dub_jobs[job_id]
   313→    with db_conn() as conn:
   314→        row = conn.execute("SELECT job_data FROM dub_history WHERE id=?", (job_id,)).fetchone()
   315→    if row and row["job_data"]:
   316→        try:
   317→            job = json.loads(row["job_data"])
   318→            with _dub_jobs_lock:
   319→                _dub_jobs[job_id] = job
   320→            return job
   321→        except json.JSONDecodeError:
   322→            # job_id arrives from request paths — strip newlines so a crafted
   323→            # id can't forge extra log lines (py/log-injection).
   324→            safe_id = str(job_id).replace("\r", "").replace("\n", "")
   325→            logger.exception("Failed to decode dub_history.job_data for %s", safe_id)
   326→    return None

   335→def merge_job(job_id: str, updates: dict) -> bool:
   336→    """Merge *updates* into an existing in-memory job. Does NOT persist.
   337→
   338→    Returns ``False`` when the job is gone — which is a real, reachable state,
   339→    not a defensive nicety: ingest runs for minutes (demucs, scene detection,
   340→    thumbnailing) and ``DELETE /dub/history/{id}`` pops the entry out from
   341→    under it. The pipeline used to finish with a bare
   342→    ``_dub_jobs[job_id].update(...)``, so deleting an in-flight dub surfaced as
   343→    the toast ``ingest: 'mgw39lx3'`` — ``str(KeyError)`` is the repr of the
   344→    key, nothing more (#1252/#1253). Callers treat ``False`` as "the user
   345→    withdrew this job" and stop, rather than resurrecting a record that was
   346→    deliberately deleted.
   347→    """
   348→    with _dub_jobs_lock:
   349→        job = _dub_jobs.get(job_id)
   350→        if job is None:
   351→            return False
   352→        job.update(updates)
   353→        return True
   354→

   356→def _expire_withdrawn(now: float, protected: int = 0) -> None:
   357→    """Drop withdrawal markers that are too old to still matter.
   358→
   359→    Caller must hold ``_dub_jobs_lock``. Age first — that is the policy — then
   360→    a size cap purely so the mapping cannot grow without bound.
   361→
   362→    ``protected`` is how many markers the current purge just recorded. Those sit
   363→    at the end (newest) and are never evicted by the size cap: they are the most
   364→    likely to still be held by a running job, and dropping one is exactly the
   365→    resurrection this whole mechanism exists to prevent. A single "clear
   366→    history" larger than the cap would otherwise force us to discard live
   367→    markers — which is what broke CI. The bound therefore is
   368→    ``cap + one purge``, not ``cap``.
   369→    """
   370→    cutoff = now - _WITHDRAWN_TTL_S
   371→    while _withdrawn_jobs:
   372→        _, deleted_at = next(iter(_withdrawn_jobs.items()))
   373→        if deleted_at >= cutoff:
   374→            break
   375→        _withdrawn_jobs.popitem(last=False)
   376→    floor = max(_WITHDRAWN_MAX, protected)
   377→    while len(_withdrawn_jobs) > floor:
   378→        _withdrawn_jobs.popitem(last=False)

   381→def begin_ingest(job_id: str) -> None:
   382→    """Mark an ingest as running.
   383→
   384→    Re-importing an id is a deliberate revival, so this clears any tombstone —
   385→    the only thing that legitimately un-deletes a job.
   386→    """
   387→    with _dub_jobs_lock:
   388→        _inflight_jobs.add(job_id)
   389→        _withdrawn_jobs.pop(job_id, None)
   390→
   391→
   392→def end_ingest(job_id: str) -> None:
   393→    """Mark an ingest as finished, however it ended.
   394→
   395→    Deliberately does NOT clear the tombstone: the ingest ending is not the
   396→    user un-deleting anything, and a render started before the delete can still
   397→    be holding that job.
   398→    """
   399→    with _dub_jobs_lock:
   400→        _inflight_jobs.discard(job_id)

   403→def put_and_save_job(
   404→    job_id: str,
   405→    job: dict,
   406→    *,
   407→    filename: str = "",
   408→    duration: float = 0.0,
   409→    content_hash: str = "",
   410→) -> bool:
   411→    """:func:`put_job` and :func:`save_job` as ONE atomic step.
   412→
   413→    Returns ``False`` when the job was withdrawn — the user deleted or cleared
   414→    its history while this ingest was running — in which case nothing is
   415→    written. Gating on the tombstone rather than on dict membership is what
   416→    makes this correct for a job's FIRST write, where an absent key is normal
   417→    (#1252 review).
   418→    """
   419→    with _dub_jobs_lock:
   420→        if job_id in _withdrawn_jobs:
   421→            return False
   422→        _dub_jobs[job_id] = job
   423→        save_job(job_id, job, filename, duration, content_hash)
   424→        return True
   425→
   426→
   427→def merge_and_save_job(
   428→    job_id: str,
   429→    updates: dict,
   430→    *,
   431→    filename: str = "",
   432→    duration: float = 0.0,
   433→    content_hash: str = "",
   434→) -> bool:
   435→    """:func:`merge_job` and :func:`save_job` as ONE atomic step.
   436→
   437→    Splitting them leaves a window that resurrects deleted work (#1252 review):
   438→    merge succeeds, the user deletes the dub — removing the row *and* the
   439→    in-memory entry — and the pending ``save_job`` then UPSERTs the row straight
   440→    back, so a dub the user deleted reappears in history. The delete endpoints
   441→    take this same lock around their own row-delete + evict, so the two
   442→    sequences cannot interleave at all.
   443→
   444→    Returns ``False`` when the job is already gone; the caller stops there.
   445→
   446→    The ``save_job`` write happens INSIDE the lock deliberately. That serialises
   447→    dub job-state access against one SQLite UPSERT — normally microseconds under
   448→    WAL, but up to sqlite3's 5 s default busy timeout if another writer is
   449→    holding the write lock. The alternative — releasing the lock before the
   450→    write — is the resurrection race this exists to close, so a rare latency
   451→    blip is the better trade. No locked region here calls another locked
   452→    function, so the plain (non-reentrant) ``_dub_jobs_lock`` cannot deadlock.
   453→    """
   454→    with _dub_jobs_lock:
   455→        job = _dub_jobs.get(job_id)
   456→        if job is None or job_id in _withdrawn_jobs:
   457→            return False
   458→        job.update(updates)
   459→        save_job(job_id, job, filename, duration, content_hash)
   460→        return True

   463→def purge_jobs(job_ids, *, delete_rows, include_inflight: bool = False) -> None:
   464→    """Delete history rows and evict the in-memory records as ONE atomic step.
   465→
   466→    ``delete_rows`` is called with the lock held, so a concurrent
   467→    :func:`merge_and_save_job` cannot slip between the row-delete and the
   468→    evict and write the job straight back (#1252 review). Both delete
   469→    endpoints go through here; ``DELETE /dub/history`` previously never evicted
   470→    from memory at all, so an in-flight job survived "clear history" entirely
   471→    and re-saved itself on completion.
   472→    """
   473→    with _dub_jobs_lock:
   474→        delete_rows()
   475→        # Deterministic order, de-duplicated. A `set` here made which markers
   476→        # the size cap evicts depend on PYTHONHASHSEED — the test for this very
   477→        # behaviour passed locally and failed in CI for that reason alone.
   478→        targets = list(dict.fromkeys(job_ids))
   479→        if include_inflight:
   480→            # "Clear history" means everything, including a job whose first row
   481→            # hasn't been written yet — it wouldn't appear in `job_ids` at all.
   482→            targets += [j for j in sorted(_inflight_jobs) if j not in set(targets)]
   483→        now = time.monotonic()
   484→        for job_id in targets:
   485→            _dub_jobs.pop(job_id, None)
   486→            _withdrawn_jobs.pop(job_id, None)
   487→            _withdrawn_jobs[job_id] = now  # most-recent last
   488→        _expire_withdrawn(now, protected=len(targets))

   491→def save_job(job_id: str, job: dict, filename: str = "", duration: float = 0.0, content_hash: str = "") -> None:
   492→    """Persist dub job state to SQLite so it survives restarts. Uses UPSERT
   493→    on `id` so repeated saves in a session keep the latest snapshot.
   494→
   495→    language / language_code / content_hash only update when the incoming
   496→    value is non-empty: the ingest-time insert runs before the target
   497→    language is known (both columns ""), generation sets them on the job
   498→    dict, and a later save from a job that lost them (e.g. hydrated from an
   499→    old row) must not clobber the healed columns back to "". The frontend
   500→    keys history restore off language_code, so a frozen "" hid finished
   501→    tracks until the user re-picked a language.
   502→    """
   503→    with _dub_jobs_lock:
   504→        # The withdrawal gate lives HERE, not in the callers (#1252 review).
   505→        # Eight call sites across generate / translate / export / core persist
   506→        # jobs directly, so gating only the ingest helpers left every
   507→        # post-ingest save able to resurrect a dub the user deleted mid-render.

   550→def run_proc_factory(job_id: str):
   551→    """Return an async `run_proc` helper bound to this job_id.
   552→
   553→    Spawns subprocesses under the shared ffmpeg semaphore, tracks them so
   554→    `kill_job_procs(job_id)` can terminate them, raises HTTP 504 on timeout.
   555→    """
   556→    async def run_proc(cmd, timeout: float = 900.0):
   557→        async with _get_semaphore():
   558→            p = await _spawn_with_retry(
   559→                cmd,
   560→                stdout=asyncio.subprocess.PIPE,
   561→                stderr=asyncio.subprocess.PIPE,
   562→            )
   563→            register_proc(job_id, p)
   564→            try:
   565→                try:
   566→                    stdout, stderr = await asyncio.wait_for(p.communicate(), timeout=timeout)
   567→                except asyncio.TimeoutError:
   568→                    try:
   569→                        p.kill()
   570→                    except ProcessLookupError:
   571→                        pass
   572→                    try:
   573→                        await asyncio.wait_for(p.wait(), timeout=5.0)
   574→                    except asyncio.TimeoutError:
   575→                        pass
   576→                    raise HTTPException(status_code=504, detail=f"subprocess timed out after {timeout}s")
   577→                return p, stdout, stderr
   578→            finally:
   579→                unregister_proc(job_id, p)
   580→                if p.returncode is None:
   581→                    try:
   582→                        p.kill()
   583→                    except ProcessLookupError:
   584→                        pass
   585→                    try:
   586→                        await asyncio.wait_for(p.wait(), timeout=5.0)
   587→                    except asyncio.TimeoutError:
   588→                        pass
   589→    return run_proc

   592→async def run_proc_streaming_stderr(
   593→    job_id: str,
   594→    cmd: list[str],
   595→    *,
   596→    timeout: float = 1800.0,
   597→) -> AsyncIterator[tuple]:
   598→    """Spawn `cmd` and stream its stderr line-by-line as it runs.
   599→
   600→    Yields:
   601→      ('stderr', line: str) — once per logical line (split on \\r or \\n,
   602→        so tqdm-style progress bars that overwrite the same line via
   603→        carriage return surface as separate lines)
   604→      ('done', returncode: int, full_stderr: bytes) — exactly once when
   605→        the subprocess exits; this is always the final value.
   606→
   607→    Honors the same semaphore + register_proc/kill plumbing as run_proc.
   608→    """
   609→    async with _get_semaphore():
   610→        p = await _spawn_with_retry(
   611→            cmd,
   612→            stdout=asyncio.subprocess.PIPE,
   613→            stderr=asyncio.subprocess.PIPE,
   614→        )
   615→        register_proc(job_id, p)
   616→        stderr_parts: list[bytes] = []
   617→        rc: int = -1
   618→        try:
   619→            if getattr(p, "uses_sync_pipes", False):
   620→                # Fallback loops (the Windows SelectorEventLoop uvicorn forces
   621→                # under --reload) hand back a thread-backed proc whose .stderr is
   622→                # a plain SYNC pipe, not an asyncio StreamReader — `await
   623→                # p.stderr.read()` there raises "a coroutine or an awaitable is
   624→                # required" and crashed the demucs step. We can't stream that
   625→                # pipe incrementally without leaking blocked executor threads on
   626→                # every 1s poll, so run to completion via the wrapper's async
   627→                # communicate() and replay stderr as the same line events. No
   628→                # live progress on that degraded loop, but the subprocess still
   629→                # runs and the emitted event sequence is identical. The native
   630→                # async path (Proactor/posix — every release build) is the
   631→                # unchanged `else` below.
   632→                try:
   633→                    _out, err_bytes = await asyncio.wait_for(
   634→                        p.communicate(), timeout=timeout
   635→                    )
   636→                except asyncio.TimeoutError:
   637→                    try:
   638→                        p.kill()
   639→                    except ProcessLookupError:
   640→                        pass
   641→                    raise HTTPException(
   642→                        status_code=504,
   643→                        detail=f"subprocess timed out after {timeout}s",
   644→                    )
   645→                err_bytes = err_bytes or b""
   646→                stderr_parts.append(err_bytes)
   647→                for _line in re.split(rb"[\r\n]", err_bytes):
   648→                    _text = _line.decode(errors="replace")
   649→                    if _text.strip():
   650→                        yield ("stderr", _text)
   651→            else:
   652→                buf = b""
   653→                start = time.monotonic()
   654→                while True:
   655→                    if time.monotonic() - start > timeout:
   656→                        try:
   657→                            p.kill()
   658→                        except ProcessLookupError:
   659→                            pass
   660→                        raise HTTPException(
   661→                            status_code=504,
   662→                            detail=f"subprocess timed out after {timeout}s",
   663→                        )
   664→                    try:
   665→                        chunk = await asyncio.wait_for(p.stderr.read(256), timeout=1.0)
   666→                    except asyncio.TimeoutError:
   667→                        if p.returncode is not None:
   668→                            break
   669→                        continue
   670→                    if not chunk:
   671→                        break
   672→                    stderr_parts.append(chunk)
   673→                    buf += chunk
   674→                    while True:
   675→                        idx_r = buf.find(b"\r")
   676→                        idx_n = buf.find(b"\n")
   677→                        if idx_r < 0 and idx_n < 0:
   678→                            break
   679→                        if idx_r < 0:
   680→                            idx = idx_n
   681→                        elif idx_n < 0:
   682→                            idx = idx_r
   683→                        else:
   684→                            idx = min(idx_r, idx_n)
   685→                        line = buf[:idx].decode(errors="replace")
   686→                        buf = buf[idx + 1:]
   687→                        if line.strip():
   688→                            yield ("stderr", line)
   689→            rc = await p.wait()
   690→        finally:
   691→            unregister_proc(job_id, p)
   692→            if p.returncode is None:
   693→                try:
   694→                    p.kill()
   695→                except ProcessLookupError:
   696→                    pass
   697→                try:
   698→                    await asyncio.wait_for(p.wait(), timeout=5.0)
   699→                except asyncio.TimeoutError:
   700→                    pass
   701→            try:
   702→                p.stdout.close()
   703→            except Exception:
   704→                pass
   705→    yield ("done", rc, b"".join(stderr_parts))
   706→

   709→_BROWSER_AUDIO_CODECS = {"aac", "mp4a"}

   712→def _probe_codecs(path: str) -> tuple[str, str]:
   713→    """Return (video_codec, audio_codec) lowercased, or ('','') on probe error."""
   714→    ffprobe = find_ffprobe()
   715→    if not ffprobe:
   716→        return ("", "")
   717→    try:
   718→        out = subprocess.run(
   719→            [ffprobe, "-v", "error", "-select_streams", "v:0",
   720→             "-show_entries", "stream=codec_name", "-of", "csv=p=0", path],
   721→            capture_output=True, text=True, timeout=10,
   722→        )
   723→        vcodec = (out.stdout or "").strip().lower()
   724→        out = subprocess.run(
   725→            [ffprobe, "-v", "error", "-select_streams", "a:0",
   726→             "-show_entries", "stream=codec_name", "-of", "csv=p=0", path],
   727→            capture_output=True, text=True, timeout=10,
   728→        )
   729→        acodec = (out.stdout or "").strip().lower()
   730→        return vcodec, acodec
   731→    except Exception:
   732→        return ("", "")

   735→def _ensure_browser_playable_mp4(video_path: str) -> str:
   736→    """Guarantee `video_path` is an mp4 with h264 video + aac audio.
   737→
   738→    Three classes of fix the in-app `<video>` element needs:
   739→      1. `.webm`/`.mkv` containers — WKWebView refuses them outright.
   740→      2. `.mp4` with VP9 or AV1 video — WKWebView can't decode either
   741→         inside an mp4 container (Safari ships VP9 only in WebM).
   742→      3. `.mp4` with Opus audio — same story; mp4-with-opus is rare and
   743→         poorly supported across webviews.
   744→
   745→    Strategy: probe codecs; if extension is mp4 AND both codecs are in
   746→    the browser-safe set, leave alone. Otherwise transcode to h264+aac.
   747→    Returns the (possibly new) file path.
   748→    """
   749→    ffmpeg_bin = find_ffmpeg()
   750→    is_mp4 = video_path.lower().endswith(".mp4")
   751→    if is_mp4:
   752→        vcodec, acodec = _probe_codecs(video_path)
   753→        if vcodec in _BROWSER_VIDEO_CODECS and acodec in _BROWSER_AUDIO_CODECS:
   754→            return video_path  # already safe — fast path, no rewrite
   755→    # Need to rewrite. Try stream-copy first when the container is the
   756→    # only problem; fall back to full transcode for codec mismatches.
   757→    target = os.path.splitext(video_path)[0] + ".mp4"
   758→    if target == video_path:
   759→        target = os.path.splitext(video_path)[0] + ".browser.mp4"
   760→    # Stream-copy attempt — works when codecs are already h264+aac but
   761→    # the container is wrong (rare with our format selector but cheap to
   762→    # try). Skip straight to transcode if we already know codecs are bad.
   763→    rc = 1
   764→    if is_mp4:
   765→        # Codecs were probed and known-bad; skip the copy attempt.
   766→        pass
   767→    else:
   768→        rc = subprocess.run(
   769→            [ffmpeg_bin, "-y", "-i", video_path,
   770→             "-c:v", "copy", "-c:a", "copy",
   771→             "-movflags", "+faststart", target],
   772→            capture_output=True,
   773→        ).returncode
   774→        if rc != 0 or not os.path.exists(target):
   775→            rc = 1
   776→    if rc != 0:
   777→        # Full transcode — h264 baseline-ish + aac is the safe combo.
   778→        rc = subprocess.run(
   779→            [ffmpeg_bin, "-y", "-i", video_path,
   780→             "-c:v", "libx264", "-preset", "veryfast", "-crf", "23",
   781→             "-pix_fmt", "yuv420p",
   782→             "-c:a", "aac", "-b:a", "192k",
   783→             "-movflags", "+faststart", target],
   784→            capture_output=True,
   785→        ).returncode
   786→    if rc == 0 and os.path.exists(target) and target != video_path:
   787→        try:
   788→            os.remove(video_path)
   789→        except OSError:
   790→            pass
   791→        return target
   792→    if rc != 0:
   793→        logger.warning(
   794→            "Could not transcode %s to browser-playable mp4 — the in-app "
   795→            "video player may render this file as a black box.",
   796→            log_safe(video_path),
   797→        )
   798→    return video_path

   801→async def _ensure_browser_playable_mp4_for_job(job_id: str, video_path: str) -> str:
   802→    """Normalize an upload through the job's cancellable process registry."""
   803→    is_mp4 = video_path.lower().endswith(".mp4")
   804→    vcodec, acodec = await asyncio.to_thread(_probe_codecs, video_path)
   805→    if is_mp4 and vcodec in _BROWSER_VIDEO_CODECS and acodec in _BROWSER_AUDIO_CODECS:
   806→        return video_path
   807→
   808→    target = os.path.splitext(video_path)[0] + ".mp4"
   809→    if target == video_path:
   810→        target = os.path.splitext(video_path)[0] + ".browser.mp4"
   811→    run_proc = run_proc_factory(job_id)
   812→    ffmpeg_bin = find_ffmpeg()
   813→
   814→    async def attempt(cmd: list[str]) -> int:
   815→        try:
   816→            proc, _stdout, _stderr = await run_proc(cmd, timeout=1800.0)
   817→            return proc.returncode
   818→        except asyncio.CancelledError:
   819→            raise
   820→        except Exception as exc:
   821→            logger.warning(
   822→                "Browser-media normalization process failed for %s: %s",
   823→                log_safe(video_path),
   824→                log_safe(exc),
   825→            )
   826→            return 1
   827→
   828→    rc = 1
   829→    if not is_mp4:
   830→        rc = await attempt(
   831→            [
   832→                ffmpeg_bin, "-y", "-i", video_path,
   833→                "-c:v", "copy", "-c:a", "copy",
   834→                "-movflags", "+faststart", target,
   835→            ]
   836→        )
   837→        if rc == 0 and os.path.exists(target):
   838→            target_vcodec, target_acodec = await asyncio.to_thread(_probe_codecs, target)
   839→            if (
   840→                target_vcodec not in _BROWSER_VIDEO_CODECS
   841→                or target_acodec not in _BROWSER_AUDIO_CODECS
   842→            ):
   843→                rc = 1
   844→        else:
   845→            rc = 1
   846→    if rc != 0:
   847→        rc = await attempt(
   848→            [
   849→                ffmpeg_bin, "-y", "-i", video_path,
   850→                "-c:v", "libx264", "-preset", "veryfast", "-crf", "23",
   851→                "-pix_fmt", "yuv420p", "-c:a", "aac", "-b:a", "192k",
   852→                "-movflags", "+faststart", target,
   853→            ]
   854→        )
   855→    if rc == 0 and os.path.exists(target) and target != video_path:
   856→        try:
   857→            os.remove(video_path)
   858→        except OSError:
   859→            pass  # Best effort: the normalized target is already complete.
   860→        return target
   861→    logger.warning(
   862→        "Could not transcode %s to browser-playable mp4 — the in-app "
   863→        "video player may render this file as a black box.",
   864→        log_safe(video_path),
   865→    )
   866→    return video_path

   869→# Bounded retry for transient download failures (#579/#598). yt-dlp's own
   870→# `retries`/`fragment_retries` cover per-fragment HTTP flakes, but a broken
   871→# pipe ([Errno 32]) raised while the write side of a pipe closes mid-stream
   872→# (a killed ffmpeg merge child, a CDN reset during muxing) aborts the whole
   873→# `extract_info` call and is NOT covered by them — so a single transient blip
   874→# failed the entire ingest with a raw "Broken pipe". We add a small download-
   875→# level retry on top, cleaning up the partial download between attempts so a
   876→# half-written `original.*` can't poison the next try.
   877→_YT_DOWNLOAD_RETRIES = 2  # total attempts = 1 + retries = 3

   880→def _with_target_facts(exc: BaseException, job_dir: str) -> BaseException:
   881→    """``exc`` with the download destination described, when the failure looks
   882→    like the OS refusing a file operation (#1225).
   883→
   884→    Returns ``exc`` untouched for network/format failures — their message is
   885→    already about the remote side, and appending disk facts would just be
   886→    noise. Never raises."""
   887→    try:
   888→        # Shared with failure.classify() so the "is this a disk problem?"
   889→        # answer can't differ between the class we assign and whether we
   890→        # bother naming the folder (#1225 review).
   891→        if not failure.is_os_write_refusal(str(exc)):
   892→            return exc
   893→        facts = failure.describe_path_target(os.path.join(job_dir, "original.mp4"))
   894→        if not facts:
   895→            return exc
   896→        msg = (
   897→            f"{exc} — saving to {job_dir} ({facts}). The OS refused the write, "
   898→            f"so retrying the same link won't help: check the drive isn't full, "
   899→            f"the folder is writable, and antivirus or a cloud-sync client "
   900→            f"(OneDrive, Dropbox) isn't locking it."
   901→        )
   902→        try:
   903→            return type(exc)(msg)
   904→        except Exception:
   905→            # Not every exception class takes a plain message (soundfile's
   906→            # LibsndfileError wants an int code). Keep the text, drop the type.
   907→            return RuntimeError(msg)
   908→    except Exception:
   909→        return exc

   912→def _is_transient_download_error(exc: BaseException) -> bool:
   913→    """True when a download failure is worth retrying (broken pipe / net drop).
   914→
   915→    Reuses the single failure taxonomy (`VIDEO_DOWNLOAD_NETWORK`) rather than a
   916→    parallel keyword list, so "what counts as transient" stays single-sourced
   917→    with the error-hint classification. ``BrokenPipeError``/``ConnectionError``
   918→    are matched by class too, since a bare instance may be wrapped or re-raised
   919→    with a stripped message that no longer contains "broken pipe".
   920→    """
   921→    if isinstance(exc, (BrokenPipeError, ConnectionError)):
   922→        return True
   923→    return failure.classify(str(exc)) == "VIDEO_DOWNLOAD_NETWORK"

   933→def _is_forbidden_download_error(exc: BaseException) -> bool:
   934→    """True for a failure the CURRENT player client can't get past, but another
   935→    one commonly can.
   936→
   937→    A 403 is the original case (#625): extraction worked, the media fetch was
   938→    refused, and the same client keeps refusing. "This video is DRM protected"
   939→    (#1254) behaves identically and belongs here for the same reason — YouTube

   941→    not actually DRM'd. The reporter saw it fail and then succeed on a plain
   942→    retry of the same URL, which is exactly what a per-client format set looks
   943→    like from outside. Escalating the client is the fix; a bare retry only
   944→    works when the next attempt happens to draw a different one.

   946→    s = str(exc)
   947→    if "403" in s or "Forbidden" in s:
   948→        return True
   949→    low = s.lower()
   950→    return "drm protected" in low or "drm-protected" in low

  1002→    url: str,
  1003→    job_dir: str,

  1007→    progress_hook=None,
  1008→    cookie_file: str | None = None,
  1009→) -> tuple[str, str, list[str]]:
  1010→    """Blocking yt-dlp download into `job_dir`.
  1011→
  1012→    Returns (video_path, title, downloaded_sub_files).
  1013→
  1014→    Captions are downloaded in a separate, best-effort pass after the video
  1015→    so subtitle failures (rate-limits, missing tracks) can never derail the
  1016→    actual ingest. Defaults skip YouTube's auto-translated set — requesting
  1017→    "all" expands into ~100 per-language variants per video and reliably
  1018→    trips HTTP 429, and the downstream Translate step handles target
  1019→    languages more reliably than YouTube's machine translations anyway.
  1020→    Pass an explicit `sub_langs` list to override the default selection.
  1021→    """
  1022→    import glob
  1023→    import yt_dlp
  1024→    outtmpl = os.path.join(job_dir, "original.%(ext)s")
  1025→    # #1225: yt-dlp surfaces an OS write rejection as a bare
  1026→    # "Unable to download video: [Errno 22] Invalid argument" — no path, no
  1027→    # reason, and three manual retries all fail identically because nothing
  1028→    # about it is transient. Fail here instead, naming the directory, when we
  1029→    # can already see it won't work.
  1030→    _target_facts = failure.describe_path_target(outtmpl)
  1031→    if "not writable" in _target_facts or "does not exist" in _target_facts:
  1032→        # Worded so classify() places it in the download path: it must carry
  1033→        # both an OS-refusal signature and download context, or the user gets
  1034→        # no hint at all — the failure this PR exists to fix (#1225 review).
  1035→        raise OSError(
  1036→            f"Unable to download video: unable to open for writing in "
  1037→            f"{job_dir} ({_target_facts}). The video downloads into this job "
  1038→            f"folder under your VoiceStudio data directory — check it exists, is "
  1039→            f"writable, and isn't locked by antivirus or a cloud-sync client."
  1040→        )
  1041→    ydl_opts: dict = {
  1042→        "outtmpl": outtmpl,
  1043→        # Prefer h264+aac streams so the merged mp4 is natively decodable
  1044→        # by WKWebView/WebView2/Chromium. YouTube serves VP9+Opus as the
  1045→        # default high-quality combo, which yt-dlp will happily mux into
  1046→        # an mp4 container — but Safari/WKWebView refuses to decode VP9
  1047→        # inside mp4, leaving a black <video> in the dub editor. Fall
  1048→        # back to any combo only when no h264/aac variant exists, then
  1049→        # the post-download codec probe below will transcode it.
  1050→        "format": (
  1051→            "bv*[vcodec^=avc1][ext=mp4]+ba[acodec^=mp4a][ext=m4a]/"
  1052→            "bv*[vcodec^=avc1]+ba[acodec^=mp4a]/"
  1053→            "b[vcodec^=avc1][acodec^=mp4a]/"
  1054→            "bv*[vcodec^=avc1]+ba/"
  1055→            "b[vcodec^=avc1]/"
  1056→            "bv*+ba/b"
  1057→        ),
  1058→        "merge_output_format": "mp4",
  1059→        "noplaylist": True,
  1060→        "quiet": True,
  1061→        "no_warnings": True,
  1062→        # `quiet` alone still lets yt-dlp draw its progress bar on stdout and
  1063→        # print errors to stderr. In the packaged app those are dead or closed
  1064→        # pipes (Windows `[Errno 22] Invalid argument`, macOS `[Errno 32]
  1065→        # Broken pipe`), which fails a download that was otherwise fine. Send
  1066→        # every yt-dlp message to our logger and never write progress to stdio.
  1067→        "noprogress": True,
  1068→        "logger": _YtdlpLogger(),
  1069→        "restrictfilenames": True,
  1070→        # Don't stamp the downloaded file's mtime with the video's upload date
  1071→        # (#642): on Windows an out-of-range/invalid timestamp makes the os.utime
  1072→        # call raise `[Errno 22] Invalid argument`, failing the whole ingest. We
  1073→        # download to a throwaway `original.*` and never use its mtime, so skip
  1074→        # it entirely (equivalent to yt-dlp's --no-mtime).
  1075→        "updatetime": False,
  1076→        "socket_timeout": 30,
  1077→        # Resilience against YouTube CDN flakes: a single empty fragment
  1078→        # (commonly the very last one — "Did not get any data blocks")
  1079→        # used to fail the whole ingest at 99% complete. Retry each
  1080→        # fragment generously and tolerate one that never returns data;
  1081→        # missing the last <0.5% of audio is acceptable for transcription.
  1082→        "fragment_retries": 10,
  1083→        "retries": 10,
  1084→        "extractor_retries": 5,
  1085→        "skip_unavailable_fragments": True,
  1086→    }
  1087→    if cookie_file:
  1088→        ydl_opts["cookiefile"] = cookie_file
  1089→    # #712: the format selector above pulls separate video+audio streams, so
  1090→    # yt-dlp muxes them via ffmpeg (merge_output_format=mp4). yt-dlp only looks
  1091→    # for ffmpeg on PATH and aborts with "you have requested merging of multiple
  1092→    # formats but ffmpeg is not installed" — but VoiceStudio's ffmpeg is often a
  1093→    # bundled Tauri sidecar / imageio-ffmpeg binary that isn't on PATH (common on
  1094→    # Windows). Point yt-dlp at the exact ffmpeg we resolve so the merge works.
  1095→    _ffmpeg_bin = find_ffmpeg()
  1096→    if _ffmpeg_bin:
  1097→        ydl_opts["ffmpeg_location"] = _ffmpeg_bin
  1098→    if progress_hook is not None:
  1099→        ydl_opts["progress_hooks"] = [progress_hook]
  1100→
  1101→    # Download with a bounded retry on transient/broken-pipe-class failures
  1102→    # (#579/#598). A broken pipe mid-mux isn't recoverable inside yt-dlp's own
  1103→    # fragment retries, but a fresh `extract_info` usually succeeds. Between
  1104→    # attempts we wipe the partial `original.*` so a half-written file can't be
  1105→    # mistaken for a finished download.
  1106→    info = None
  1107→    path = None
  1108→    transient_used = 0
  1109→    client_idx = 0
  1110→    while True:
  1111→        try:
  1112→            with yt_dlp.YoutubeDL(ydl_opts) as ydl:
  1113→                info = ydl.extract_info(url, download=True)
  1114→                path = ydl.prepare_filename(info)
  1115→            break
  1116→        except Exception as exc:
  1117→            _cleanup_partial_download(job_dir)
  1118→            # 403 Forbidden: not transient — escalate the YouTube player client,
  1119→            # which commonly bypasses a signature-protected format set (#625).
  1120→            if _is_forbidden_download_error(exc) and client_idx < len(_YT_PLAYER_CLIENTS):
  1121→                client = _YT_PLAYER_CLIENTS[client_idx]
  1122→                client_idx += 1
  1123→                ydl_opts = {**ydl_opts, "extractor_args": {"youtube": {"player_client": [client]}}}
  1124→                logger.warning(
  1125→                    "Download 403 for %s — retrying with player_client=%s (#625)", log_safe(url), log_safe(client),
  1126→                )
  1127→                continue
  1128→            # Transient/broken-pipe: a fresh extract_info usually succeeds
  1129→            # (#579/#598). A 403 never counts here — it's escalated above.
  1130→            if (transient_used < _YT_DOWNLOAD_RETRIES
  1131→                    and _is_transient_download_error(exc)
  1132→                    and not _is_forbidden_download_error(exc)):
  1133→                transient_used += 1
  1134→                logger.warning(
  1135→                    "Transient download failure for %s (attempt %d/%d): %s — retrying",
  1136→                    log_safe(url), transient_used, _YT_DOWNLOAD_RETRIES, log_safe(exc),
  1137→                )
  1138→                time.sleep(2 * transient_used)  # brief, increasing backoff
  1139→                continue
  1140→            # #1225: an OS-level rejection (errno 22 / EACCES / ENOSPC) tells
  1141→            # the user nothing on its own. Attach what we can observe about
  1142→            # the destination so the message identifies a full drive, a
  1143→            # removed folder, or an antivirus/cloud-sync lock. Wording keeps
  1144→            # the yt-dlp text so classify() still sees the download context.
  1145→            described = _with_target_facts(exc, job_dir)
  1146→            if described is exc:
  1147→                raise
  1148→            raise described from exc
  1149→    root, _ = os.path.splitext(path)
  1150→    mp4 = root + ".mp4"
  1151→    if os.path.exists(mp4):
  1152→        video_path = mp4
  1153→    else:
  1154→        video_path = path
  1155→    # Browser-playability guard: WKWebView (Tauri on macOS) refuses to
  1156→    # decode VP9/AV1 video and Opus audio even when they're wrapped in an
  1157→    # mp4 container, and refuses .webm/.mkv outright. We probe the actual
  1158→    # codecs and transcode only when needed — `<video>` rendering is
  1159→    # what we care about, not what extension yt-dlp ended up writing.
  1160→    video_path = _ensure_browser_playable_mp4(video_path)
  1161→    title = info.get("title") or os.path.basename(video_path)
  1162→
  1163→    sub_files: list[str] = []
  1164→    if fetch_subs:
  1165→        if sub_langs:
  1166→            langs = list(sub_langs)
  1167→        else:
  1168→            langs = _default_caption_languages(info)
  1169→        if not langs:
  1170→            logger.info("No captions available on %s (skipping subtitle pass)", log_safe(url))
  1171→        else:
  1172→            sub_opts = {
  1173→                **ydl_opts,
  1174→                "skip_download": True,
  1175→                "writesubtitles": True,
  1176→                "writeautomaticsub": True,
  1177→                "subtitleslangs": langs,
  1178→                "subtitlesformat": "vtt",
  1179→                "extractor_args": {"youtube": {"skip": ["translated_subs"]}},
  1180→                "ignoreerrors": True,
  1181→                "extractor_retries": 5,
  1182→                "sleep_interval_subtitles": 1,
  1183→            }
  1184→            try:
  1185→                with yt_dlp.YoutubeDL(sub_opts) as ydl_sub:
  1186→                    ydl_sub.extract_info(url, download=True)
  1187→            except Exception as e:
  1188→                logger.warning(
  1189→                    "Subtitle download failed for %s (continuing with video): %s",
  1190→                    log_safe(url), log_safe(e),
  1191→                )
  1192→            base = os.path.splitext(video_path)[0]
  1193→            sub_files = sorted(glob.glob(base + ".*.vtt"))
  1194→    return video_path, title, sub_files

  1197→def _default_caption_languages(info: dict) -> list[str]:
  1198→    """Return original-language caption tracks without translated auto-captions.
  1199→
  1200→    Some extractors omit ``language`` even though yt-dlp exposes an original
  1201→    automatic-caption track such as ``en-orig``.  Treat that explicit suffix as
  1202→    source metadata so caption-first ingest still works instead of needlessly
  1203→    loading ASR.  Manual tracks remain eligible because they are authored source
  1204→    material and yt-dlp's ``skip=translated_subs`` guard still applies.
  1205→    """
  1206→    original = str(info.get("language") or "").strip()
  1207→    manual = {
  1208→        str(language).strip()
  1209→        for language in (info.get("subtitles") or {})
  1210→        if str(language).strip()
  1211→    }
  1212→    automatic = {
  1213→        str(language).strip()
  1214→        for language in (info.get("automatic_captions") or {})
  1215→        if str(language).strip()
  1216→    }
  1217→    selected = set(manual)
  1218→    if original:
  1219→        primary = original.split("-", 1)[0]
  1220→        has_source_manual = any(
  1221→            language == original or language.split("-", 1)[0] == primary
  1222→            for language in manual
  1223→        )
  1224→        if not has_source_manual:
  1225→            for candidate in (
  1226→                f"{original}-orig",
  1227→                f"{primary}-orig",
  1228→                original,
  1229→                primary,
  1230→            ):
  1231→                if candidate in automatic:
  1232→                    selected.add(candidate)
  1233→                    break
  1234→    else:
  1235→        selected.update(language for language in automatic if language.endswith("-orig"))
  1236→    return sorted(selected)

  1239→def parse_vtt_segments(vtt_path: str) -> list[dict]:
  1240→    """Very small WEBVTT parser → list of {start, end, text}.
  1241→
  1242→    Purpose: turn yt-dlp's downloaded caption track into the same segment
  1243→    shape the dub pipeline's transcript step produces, so the UI can seed
  1244→    its editor from YouTube captions instead of running Whisper. We don't
  1245→    care about positioning cues or styling — just the timed text.
  1246→    """
  1247→    try:
  1248→        with open(vtt_path, "r", encoding="utf-8") as f:
  1249→            raw = f.read()
  1250→    except OSError:
  1251→        return []
  1252→
  1253→    def _ts(s: str) -> float:
  1254→        # Format: HH:MM:SS.mmm or MM:SS.mmm
  1255→        parts = s.strip().split(":")
  1256→        try:
  1257→            parts = [float(p.replace(",", ".")) for p in parts]
  1258→        except ValueError:
  1259→            return 0.0
  1260→        if len(parts) == 3:
  1261→            h, m, sec = parts
  1262→        elif len(parts) == 2:
  1263→            h, m, sec = 0.0, parts[0], parts[1]
  1264→        else:
  1265→            return 0.0
  1266→        return h * 3600.0 + m * 60.0 + sec
  1267→
  1268→    segments: list[dict] = []
  1269→    text = raw.lstrip("﻿").replace("\r\n", "\n").replace("\r", "\n")
  1270→    for index, block in enumerate(webvtt_content_blocks(text, strict_blank=True)):
  1271→        lines = [ln for ln in block.split("\n") if ln.strip()]
  1272→        # Only the file's leading header line is metadata; the same words at
  1273→        # the start of a cue's dialogue are spoken (#2510).
  1274→        if index == 0 and lines and re.match(r"WEBVTT(?:[ \t]|$)", lines[0]):
  1275→            lines = lines[1:]
  1276→        if not lines:
  1277→            continue
  1278→        # Skip numeric cue ID line if present
  1279→        if "-->" not in lines[0] and len(lines) > 1:
  1280→            lines = lines[1:]
  1281→        if not lines or "-->" not in lines[0]:
  1282→            continue
  1283→        ts_line = lines[0]
  1284→        try:
  1285→            left, right = ts_line.split("-->")
  1286→            # Drop any settings after the right timestamp ("00:01.000 align:left line:0%")
  1287→            right = right.strip().split(" ")[0]
  1288→            start = _ts(left)
  1289→            end = _ts(right)
  1290→        except Exception:
  1291→            continue
  1292→        text = spoken_cue_text(" ".join(ln.strip() for ln in lines[1:]), webvtt=True)
  1293→        if text:
  1294→            segments.append({"start": start, "end": end, "text": text})
  1295→    return segments

  1298→async def ingest_pipeline(
  1299→    job_id: str,
  1300→    job_dir: str,
  1301→    source: dict,
  1302→    filename_hint: Optional[str] = None,
  1303→) -> AsyncIterator[str]:
  1304→    """Async generator: emit SSE events per processing stage.
  1305→
  1306→    Stages: download_start, download_done, extract_start, extract_done,
  1307→    demucs_start, demucs_done, scene_start, scene_done, ready, error, cancelled.
  1308→    """
  1309→    youtube_subs_by_lang: dict[str, list[dict]] = {}
  1310→    # Audio-only jobs (#119) skip scene detection + thumbnailing below; the
  1311→    # transcribe → translate → TTS core is identical.
  1312→    input_type = (source.get("input_type") or "video").lower()
  1313→    # Declare the run so a "clear history" arriving before this job's first
  1314→    # persistence can still withdraw it (#1252 review).
  1315→    begin_ingest(job_id)
  1316→    try:
  1317→        if source.get("kind") == "url":
  1318→            url = source["url"]
  1319→            fetch_subs = bool(source.get("fetch_subs"))
  1320→            sub_langs = source.get("sub_langs") or None
  1321→            cookie_file = source.get("cookie_file") or None
  1322→            yield prep_event("download_start", url=url)
  1323→            # Bridge yt-dlp's per-fragment progress callback (fires inside
  1324→            # the worker thread) into the async generator via a threadsafe
  1325→            # queue so the UI can render a real download bar.
  1326→            loop = asyncio.get_running_loop()
  1327→            dl_queue: asyncio.Queue = asyncio.Queue()
  1328→            _last_pct = -1
  1329→
  1330→            def _yt_progress(d: dict) -> None:
  1331→                nonlocal _last_pct
  1332→                status = d.get("status")
  1333→                if status == "downloading":
  1334→                    total = d.get("total_bytes") or d.get("total_bytes_estimate") or 0
  1335→                    cur = d.get("downloaded_bytes") or 0
  1336→                    if total > 0:
  1337→                        pct = max(0, min(100, int(cur * 100 / total)))
  1338→                        if pct != _last_pct:
  1339→                            _last_pct = pct
  1340→                            payload = {
  1341→                                "percent": pct,
  1342→                                "speed_bps": d.get("speed") or 0,
  1343→                                "eta_s": d.get("eta"),
  1344→                            }
  1345→                            loop.call_soon_threadsafe(dl_queue.put_nowait, payload)
  1346→
  1347→            dl_task = asyncio.create_task(asyncio.to_thread(
  1348→                yt_download_sync, url, job_dir,
  1349→                fetch_subs=fetch_subs, sub_langs=sub_langs,
  1350→                progress_hook=_yt_progress,
  1351→                cookie_file=cookie_file,
  1352→            ))
  1353→            try:
  1354→                while not dl_task.done():
  1355→                    try:
  1356→                        payload = await asyncio.wait_for(dl_queue.get(), timeout=0.5)
  1357→                        yield prep_event("download_progress", **payload)
  1358→                    except asyncio.TimeoutError:
  1359→                        continue
  1360→                # Drain any final queued events.
  1361→                while not dl_queue.empty():
  1362→                    payload = dl_queue.get_nowait()
  1363→                    yield prep_event("download_progress", **payload)
  1364→                video_path, title, sub_files = await dl_task
  1365→            except Exception as e:
  1366→                logger.error("Download failed for job %s: %s", log_safe(job_id), log_safe(e))
  1367→                if not dl_task.done():
  1368→                    dl_task.cancel()
  1369→                yield prep_event("error", **failure.build_failure(e, stage="download"))
  1370→                shutil.rmtree(job_dir, ignore_errors=True)
  1371→                return
  1372→            # yt-dlp (including its optional subtitle pass) is finished. Drop
  1373→            # the login credential before the much longer audio-prep stages.
  1374→            if _delete_cookie_export(cookie_file):
  1375→                source["cookie_file"] = None
  1376→            filename = title or os.path.basename(video_path)
  1377→            try:
  1378→                size = os.path.getsize(video_path)
  1379→            except OSError:
  1380→                size = 0
  1381→            # Parse each downloaded .<lang>.vtt into a segment list we can
  1382→            # stash on the job. The router will merge this into the job dict
  1383→            # after ingest completes so /dub/transcribe-stream (or a future
  1384→            # "use-youtube-subs" toggle) can seed segments from it.
  1385→            for sf_path in sub_files:
  1386→                base = os.path.splitext(os.path.basename(sf_path))[0]
  1387→                # "original.en" → lang "en"; fallback to the trailing token.
  1388→                lang_tag = base.rsplit(".", 1)[-1] if "." in base else "und"
  1389→                segs = parse_vtt_segments(sf_path)
  1390→                if segs:
  1391→                    youtube_subs_by_lang[lang_tag] = segs
  1392→            yield prep_event(
  1393→                "download_done",
  1394→                title=title, size=size, filename=filename,
  1395→                youtube_subs=sorted(youtube_subs_by_lang.keys()),
  1396→            )
  1397→        else:
  1398→            video_path = source["path"]
  1399→            filename = filename_hint or os.path.basename(video_path)
  1400→
  1401→        audio_path = os.path.join(job_dir, "audio.wav")
  1402→        audio_hq_path = os.path.join(job_dir, "audio_hq.wav")
  1403→        attempt = uuid.uuid4().hex
  1404→        extract_path = os.path.join(job_dir, f"audio-{attempt}.partial.wav")
  1405→        extract_hq_path = os.path.join(job_dir, f"audio-hq-{attempt}.partial.wav")
  1406→        ffmpeg = find_ffmpeg()
  1407→        run_proc = run_proc_factory(job_id)
  1408→
  1409→        yield prep_event("extract_start")
  1410→        try:
  1411→            await asyncio.to_thread(validate_media_source, video_path)
  1412→            # A video with no audio stream has nothing to transcribe or dub.
  1413→            # Name that instead of letting ffmpeg fail with exit 234 and a
  1414→            # stream dump ending in "Invalid argument".
  1415→            await asyncio.to_thread(require_audio_stream, video_path)
  1416→            p, _, stderr = await run_proc([
  1417→                ffmpeg, "-i", video_path, "-vn", "-acodec", "pcm_s16le",
  1418→                "-ar", "16000", "-ac", "1", extract_path, "-y",
  1419→            ])
  1420→            if p.returncode != 0:
  1421→                # The probe can be undetermined (no ffprobe); recognize the
  1422→                # same cause from ffmpeg's own wording.
  1423→                await asyncio.to_thread(raise_for_audio_extract_failure, stderr, video_path)
  1424→                msg = _media_process_error("FFmpeg", p.returncode, stderr, paths=(video_path, audio_path, job_dir))
  1425→                raise Exception(msg)
  1426→            os.replace(extract_path, audio_path)
  1427→            # Second, FULL-QUALITY extraction for source separation. audio.wav
  1428→            # is deliberately 16 kHz mono — that's what ASR wants — but Demucs
  1429→            # used to separate that same file, so the music bed inherited mono
  1430→            # (stereo image destroyed: L/R correlation 1.000 vs the original's
  1431→            # 0.754, measured) and an 8 kHz ceiling (nothing real above half
  1432→            # the ASR rate — the bed's "muffled" sound at its source). Demucs
  1433→            # resamples to 44.1 kHz internally either way, so separating the
  1434→            # stereo original costs about the same and returns a true-stereo,
  1435→            # full-band bed. Best-effort: on failure Demucs falls back to the
  1436→            # ASR file, which is exactly the old behavior.
  1437→            try:
  1438→                p_hq, _, stderr_hq = await run_proc([
  1439→                    ffmpeg, "-i", video_path, "-vn", "-acodec", "pcm_s16le",
  1440→                    "-ar", "44100", "-ac", "2", extract_hq_path, "-y",
  1441→                ])
  1442→                if p_hq.returncode != 0 or not os.path.exists(extract_hq_path):
  1443→                    logger.warning(
  1444→                        "HQ audio extraction failed (rc=%s) — separation falls "
  1445→                        "back to the 16k mono ASR file", p_hq.returncode,
  1446→                    )
  1447→                    _discard_partial_audio(extract_hq_path)
  1448→                    audio_hq_path = None
  1449→                else:
  1450→                    os.replace(extract_hq_path, audio_hq_path)
  1451→            except Exception as e_hq:  # noqa: BLE001 — quality upgrade, never fatal
  1452→                logger.warning("HQ audio extraction errored (%s) — falling back", type(e_hq).__name__)
  1453→                _discard_partial_audio(extract_hq_path)
  1454→                audio_hq_path = None
  1455→        except asyncio.CancelledError:
  1456→            _discard_partial_audio(extract_path, extract_hq_path)
  1457→            raise
  1458→        except Exception as e:
  1459→            _discard_partial_audio(extract_path, extract_hq_path)
  1460→            logger.error("Extract failed for job %s: %s", log_safe(job_id), type(e).__name__)
  1461→            if isinstance(e, failure.InvalidMediaFileError):
  1462→                await asyncio.to_thread(_discard_invalid_source_copy, job_dir, video_path)
  1463→            yield prep_event("error", **failure.build_failure(e, stage="extract"))
