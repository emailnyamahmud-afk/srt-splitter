    57→def load_env_file():
    58→    """Load .env file dari ~/Dubbing/.env (kalau ada).
    59→    Supaya user tidak perlu set env vars manual tiap kali update kamus-tui.py.
    60→    Format .env:
    61→      NEXT_PUBLIC_SUPABASE_URL=https://xxx.supabase.co
    62→      NEXT_PUBLIC_SUPABASE_ANON_KEY=eyJxxx...
    63→    """
    64→    if not ENV_FILE.exists():
    65→        return False
    66→    try:
    67→        with open(ENV_FILE, 'r', encoding='utf-8') as f:
    68→            for line in f:
    69→                line = line.strip()
    70→                # Skip comment dan empty line
    71→                if not line or line.startswith('#'):
    72→                    continue
    73→                # Parse KEY=VALUE
    74→                if '=' in line:
    75→                    key, value = line.split('=', 1)
    76→                    key = key.strip()
    77→                    value = value.strip().strip('"').strip("'")
    78→                    # Hanya set kalau belum ada di os.environ (os.environ lebih prioritas)
    79→                    if key and key not in os.environ:
    80→                        os.environ[key] = value
    81→        return True
    82→    except Exception as e:
    83→        print(f'  ⚠ Gagal load .env: {e}')
    84→        return False

   115→def save_kamus(data):
   116→    """Save ke JSON file + recompute status untuk SEMUA entries.
   117→
   118→    Status logic:
   119→    - 'ready' = ngoko + krama + arti SEMUA terisi → siap upload ke Supabase
   120→    - 'draft' = belum lengkap
   121→
   122→    SELALU recompute status setiap save (jangan skip yang sudah punya status).
   123→    User edit arti → status harus recompute dari draft → ready.
   124→    """
   125→    for w in data['words']:
   126→        ngoko = (w.get('ngoko') or '').strip()
   127→        krama = (w.get('krama') or '').strip()
   128→        arti = (w.get('arti') or '').strip()
   129→        w['status'] = 'ready' if (ngoko and krama and arti) else 'draft'
   130→    path = get_kamus_path()
   131→    with open(path, 'w', encoding='utf-8') as f:
   132→        json.dump(data, f, ensure_ascii=False, indent=2)

   293→def browse_list(data, entries_with_idx, title):
   294→    """Browse list of entries (idx, entry) dengan pagination"""
   295→    total = len(entries_with_idx)
   296→    if total == 0:
   297→        print(f'\n  Tidak ada entri untuk "{title}"')
   298→        input('\n  Tekan Enter...')
   299→        return
   300→
   301→    page_size = 20
   302→    current_page = 0
   303→
   304→    while True:
   305→        start = current_page * page_size
   306→        end = min(start + page_size, total)
   307→        page = entries_with_idx[start:end]
   308→
   309→        os.system('clear' if os.name != 'nt' else 'cls')
   310→        print('╔' + '═' * 60 + '╗')
   311→        print(f'║  {title[:58]}' + ' ' * max(0, 58 - len(title)) + '║')
   312→        print(f'║  {total} entri | Hal {current_page + 1}/{(total + page_size - 1) // page_size}' + ' ' * max(0, 30 - len(str(total)) - len(str(current_page + 1)) - len(str((total + page_size - 1) // page_size))) + '║')
   313→        print('╚' + '═' * 60 + '╝')
   314→        print()
   315→
   316→        # Build label → idx mapping (untuk lookup saat select)
   317→        label_to_idx = {}
   318→        choices = []
   319→        for orig_idx, entry in page:
   320→            ngoko = entry.get('ngoko', '') or entry.get('krama', '') or ''
   321→            ngoko_display = ngoko[:25]
   322→            krama = entry.get('krama', '') or ''
   323→            ki = entry.get('krama_inggil', '') or ''
   324→            arti = entry.get('arti', '') or ''
   325→            status = entry.get('status', 'draft')
   326→            register = entry.get('register', 'umum')
   327→            entry_id = entry.get('entry_id', orig_idx + 1)
   328→            icon = '✓' if status == 'ready' else '○'
   329→
   330→            # Build unique label — tampilkan entry_id (dari JSON) supaya user bisa referensi
   331→            label = f'{icon} #{entry_id:5d}. {ngoko_display:25s}'
   332→            if krama:
   333→                label += f' → {krama[:15]:15s}'
   334→            if ki:
   335→                label += f' | ki: {ki[:10]}'
   336→            if arti:
   337→                label += f' | {arti[:15]}'
   338→            # Tampilkan register kalau bukan default (umum)
   339→            if register != 'umum':
   340→                label += f' [{register}]'
   341→            label_to_idx[label] = orig_idx
   342→            choices.append(label)
   343→
   344→        # Navigation
   345→        nav = []
   346→        if current_page > 0:
   347→            nav.append('◀ Halaman sebelumnya')
   348→        if end < total:
   349→            nav.append('▶ Halaman berikutnya')
   350→        nav.append('🔍 Search di list ini')
   351→        nav.append('↩ Kembali ke menu utama')
   352→
   353→        choices.extend(nav)
   354→
   355→        selected = questionary.select(
   356→            f'Pilih entri (✓=sudah diedit, ○=belum):',
   357→            choices=choices,
   358→            default=choices[0],
   359→        ).ask()
   360→
   361→        if not selected or 'Kembali ke menu' in selected:
   362→            return
   363→
   364→        if 'sebelumnya' in selected:
   365→            current_page = max(0, current_page - 1)
   366→            continue
   367→
   368→        if 'berikutnya' in selected:
   369→            current_page += 1
   370→            continue
   371→
   372→        if 'Search' in selected:
   373→            search_query = questionary.text('Cari kata di list ini:').ask()
   374→            if search_query:
   375→                search_lower = search_query.lower().strip()
   376→                # Exact match dulu, baru substring (sama seperti search_menu)
   377→                exact_f = []
   378→                substr_f = []
   379→                for i, e in entries_with_idx:
   380→                    ngoko = (e.get('ngoko', '') or '').lower()
   381→                    krama = (e.get('krama', '') or '').lower()
   382→                    ki = (e.get('krama_inggil', '') or '').lower()
   383→                    arti = (e.get('arti', '') or '').lower()
   384→                    if ngoko == search_lower or krama == search_lower or ki == search_lower or arti == search_lower:
   385→                        exact_f.append((i, e))
   386→                    elif (search_lower in ngoko or search_lower in krama or
   387→                          search_lower in ki or search_lower in arti):
   388→                        substr_f.append((i, e))
   389→                filtered = exact_f + substr_f
   390→                if filtered:
   391→                    browse_list(data, filtered, f'🔍 Search "{search_query}"')
   392→                else:
   393→                    print(f'\n  Tidak ada hasil untuk "{search_query}"')
   394→                    input('\n  Tekan Enter...')
   395→            continue
   396→
   397→        # Find which entry was selected (lookup di label_to_idx)
   398→        match_idx = label_to_idx.get(selected)
   399→        if match_idx is not None:
   400→            edit_entry(data, match_idx)

   403→def search_menu(data):
   404→    """Search kamus — exact match dulu, baru substring. Hanya di ngoko/krama/krama_inggil/arti."""
   405→    os.system('clear' if os.name != 'nt' else 'cls')
   406→    print('╔' + '═' * 60 + '╗')
   407→    print('║  🔍 Search Kamus' + ' ' * 44 + '║')
   408→    print('╚' + '═' * 60 + '╝')
   409→    print()
   410→
   411→    query = questionary.text('Cari kata (di ngoko / krama / arti):').ask()
   412→    if not query:
   413→        return
   414→
   415→    search_lower = query.lower().strip()
   416→    # Exact match dulu (ngoko/krama/krama_inggil/arti persis = kata), baru substring
   417→    # TIDAK search di keterangan (terlalu banyak noise)
   418→    exact = []
   419→    substr = []
   420→    for i, entry in enumerate(data['words']):
   421→        ngoko = (entry.get('ngoko') or '').lower()
   422→        krama = (entry.get('krama') or '').lower()
   423→        ki = (entry.get('krama_inggil') or '').lower()
   424→        arti = (entry.get('arti') or '').lower()
   425→        if ngoko == search_lower or krama == search_lower or ki == search_lower or arti == search_lower:
   426→            exact.append((i, entry))
   427→        elif (search_lower in ngoko or search_lower in krama or
   428→              search_lower in ki or search_lower in arti):
   429→            substr.append((i, entry))
   430→    matches = exact + substr
   431→
   432→    if not matches:
   433→        print(f'\n  Tidak ada hasil untuk "{query}"')
   434→        input('\n  Tekan Enter...')
   435→        return
   436→
   437→    browse_list(data, matches, f'🔍 Search "{query}"')
