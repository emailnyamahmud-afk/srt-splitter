    81→    81→def parse_srt(content: str):
    82→    82→    """Parse SRT string jadi list of {index, start, end, text}."""
    83→    83→    normalized = content.replace('\r\n', '\n').replace('\r', '\n').strip()
    84→    84→    if not normalized:
    85→    85→        return []
    86→    86→    blocks = re.split(r'\n\s*\n', normalized)
    87→    87→    entries = []
    88→    88→    for block in blocks:
    89→    89→        lines = [l for l in block.split('\n') if l.strip()]
    90→    90→        if not lines:
    91→    91→            continue
    92→    92→        time_idx = -1
    93→    93→        for i, line in enumerate(lines):
    94→    94→            if '-->' in line:
    95→    95→                time_idx = i
    96→    96→                break
    97→    97→        if time_idx == -1:
    98→    98→            continue
    99→    99→        m = re.match(
   100→   100→            r'(\d{1,2}:\d{2}:\d{2}[,.]\d{1,3})\s*-->\s*(\d{1,2}:\d{2}:\d{2}[,.]\d{1,3})',
   101→   101→            lines[time_idx],
   102→   102→        )
   103→   103→        if not m:
   104→   104→            continue
   105→   105→        start = parse_time(m.group(1))
   106→   106→        end = parse_time(m.group(2))
   107→   107→        text = '\n'.join(lines[time_idx + 1:])
   108→   108→        entries.append({
   109→   109→            'index': len(entries) + 1,
   110→   110→            'start': start,
   111→   111→            'end': end,
   112→   112→            'text': text,
   113→   113→        })
   114→   114→    return entries
   115→   115→
   116→   116→
   117→   117→def parse_time(ts: str) -> float:
   118→   118→    """Parse 'HH:MM:SS,mmm' → seconds (float)."""
   119→   119→    ts = ts.replace('.', ',')
   120→   120→    hms, ms = ts.split(',') if ',' in ts else (ts, '0')
   121→   121→    h, m, s = hms.split(':')
   122→   122→    return int(h) * 3600 + int(m) * 60 + int(s) + int(ms) / 1000
   123→   123→
   124→   124→
   125→   125→def format_time_ffmpeg(seconds: float) -> str:
   126→   126→    """Format seconds → 'HH:MM:SS.mmm' (untuk ffmpeg trim)."""
   127→   127→    if seconds < 0:
   128→   128→        seconds = 0
   129→   129→    h = int(seconds // 3600)
   130→   130→    m = int((seconds % 3600) // 60)
   131→   131→    s = int(seconds % 60)
   132→   132→    ms = int((seconds - int(seconds)) * 1000)
   133→   133→    return f'{h:02d}:{m:02d}:{s:02d}.{ms:03d}'

   193→   193→
   194→   194→def build_segment_tasks(mandarin_entries, jawa_entries, mp4_duration):
   195→   195→    """
   196→   196→    Build list of segment tasks untuk Pass 1.
   197→   197→    Setiap task = 1 FFmpeg command untuk render 1 segment (cue atau gap).
   198→   198→
   199→   199→    Returns:
   200→   200→      list of dict: {
   201→   201→        'type': 'cue' | 'gap' | 'tail',
   202→   202→        'index': int,
   203→   203→        'mp4_start': float,  # start di MP4 ori
   204→   204→        'mp4_end': float,    # end di MP4 ori
   205→   205→        'target_duration': float,  # durasi setelah retim
   206→   206→        'factor': float,     # target_duration / (mp4_end - mp4_start)
   207→   207→        'output_file': str,  # path ke segment file
   208→   208→      }
   209→   209→    """
   210→   210→    tasks = []
   211→   211→    jawa_by_idx = {i: e for i, e in enumerate(jawa_entries)}
   212→   212→    last_end = 0.0
   213→   213→    seg_idx = 0
   214→   214→    cumulative_offset = 0.0  # sum durasi semua segments sebelumnya (untuk timestamp akumulatif)
   215→   215→
   216→   216→    # Filter: hanya process cues yang mp4_start < mp4_duration
   217→   217→    for i, m_entry in enumerate(mandarin_entries):
   218→   218→        if m_entry['start'] >= mp4_duration:
   219→   219→            print(f'  → Stop di cue {i} (mp4_start={m_entry["start"]:.2f}s ≥ mp4_duration={mp4_duration:.2f}s)')
   220→   220→            break
   221→   221→
   222→   222→        # Gap sebelum cue ini
   223→   223→        gap_mp4_dur = m_entry['start'] - last_end
   224→   224→        if gap_mp4_dur > 0.01:
   225→   225→            if i in jawa_by_idx and (i - 1) in jawa_by_idx:
   226→   226→                jawa_gap_dur = jawa_by_idx[i]['start'] - jawa_by_idx[i - 1]['end']
   227→   227→            elif i == 0 and i in jawa_by_idx:
   228→   228→                jawa_gap_dur = jawa_by_idx[i]['start']
   229→   229→            else:
   230→   230→                jawa_gap_dur = gap_mp4_dur
   231→   231→
   232→   232→            jawa_gap_dur = max(0.05, jawa_gap_dur)
   233→   233→            factor = jawa_gap_dur / gap_mp4_dur
   234→   234→
   235→   235→            gap_end = min(m_entry['start'], mp4_duration)
   236→   236→            tasks.append({
   237→   237→                'type': 'gap',
   238→   238→                'index': seg_idx,
   239→   239→                'mp4_start': last_end,
   240→   240→                'mp4_end': gap_end,
   241→   241→                'target_duration': jawa_gap_dur,
   242→   242→                'factor': factor,
   243→   243→                'cumulative_offset': cumulative_offset,
   244→   244→            })
   245→   245→            cumulative_offset += jawa_gap_dur
   246→   246→            seg_idx += 1
   247→   247→
   248→   248→        # Cue itu sendiri
   249→   249→        if i in jawa_by_idx and m_entry['end'] <= mp4_duration:
   250→   250→            j_entry = jawa_by_idx[i]
   251→   251→            cue_mp4_dur = m_entry['end'] - m_entry['start']
   252→   252→            cue_jawa_dur = j_entry['end'] - j_entry['start']
   253→   253→
   254→   254→            if cue_mp4_dur > 0.01 and cue_jawa_dur > 0.01:
   255→   255→                factor = cue_jawa_dur / cue_mp4_dur
   256→   256→                tasks.append({
   257→   257→                    'type': 'cue',
   258→   258→                    'index': seg_idx,
   259→   259→                    'mp4_start': m_entry['start'],
   260→   260→                    'mp4_end': m_entry['end'],
   261→   261→                    'target_duration': cue_jawa_dur,
   262→   262→                    'factor': factor,
   263→   263→                    'cumulative_offset': cumulative_offset,
   264→   264→                })
   265→   265→                cumulative_offset += cue_jawa_dur
   266→   266→                seg_idx += 1
   267→   267→
   268→   268→        last_end = min(m_entry['end'], mp4_duration)
   269→   269→
   270→   270→    # Tail gap
   271→   271→    if last_end < mp4_duration - 0.05:
   272→   272→        tail_dur = mp4_duration - last_end
   273→   273→        tasks.append({
   274→   274→            'type': 'tail',
   275→   275→            'index': seg_idx,
   276→   276→            'mp4_start': last_end,
   277→   277→            'mp4_end': mp4_duration,
   278→   278→            'target_duration': tail_dur,
   279→   279→            'factor': 1.0,
   280→   280→            'cumulative_offset': cumulative_offset,
   281→   281→        })
   282→   282→        seg_idx += 1
   283→   283→
   284→   284→    return tasks
   285→   285→

   287→   287→# ============================================================
   288→   288→# Render 1 segment (worker function untuk parallel)

   291→   291→def render_segment(task, mp4_path, segments_dir, ffmpeg_path, ffprobe_path, preset, has_audio_ori, encoder='libx264'):
   292→   292→    """
   293→   293→    Render 1 segment jadi file MP4 kecil.
   294→   294→    Dipanggil oleh ProcessPoolExecutor (parallel).
   295→   295→    """
   296→   296→    seg_idx = task['index']
   297→   297→    output_file = os.path.join(segments_dir, f'seg_{seg_idx:05d}.mp4')
   298→   298→
   299→   299→    mp4_start = task['mp4_start']
   300→   300→    mp4_end = task['mp4_end']
   301→   301→    mp4_dur = mp4_end - mp4_start
   302→   302→    target_dur = task['target_duration']
   303→   303→    factor = task['factor']
   304→   304→    cumulative_offset = task.get('cumulative_offset', 0.0)
   305→   305→
   306→   306→    # Skip kalau sudah ada (resume support)
   307→   307→    if os.path.isfile(output_file) and os.path.getsize(output_file) > 1000:
   308→   308→        return {'index': seg_idx, 'output_file': output_file, 'status': 'skipped'}
   309→   309→
   310→   310→    # Encoder params: libx264 (software) atau h264_videotoolbox (hardware M1)
   311→   311→    if encoder == 'h264_videotoolbox':
   312→   312→        enc_params = ['-c:v', 'h264_videotoolbox', '-b:v', '5M', '-realtime', '0',
   313→   313→                       '-bf', '0', '-profile:v', 'high']
   314→   314→    else:
   315→   315→        enc_params = ['-c:v', 'libx264', '-preset', preset, '-crf', '23',
   316→   316→                       '-bf', '0', '-profile:v', 'high']
   317→   317→
   318→   318→    use_stream_copy = 0.95 <= factor <= 1.05 and task['type'] in ('gap', 'tail')
   319→   319→
   320→   320→    if use_stream_copy:
   321→   321→        # Stream copy dengan output_ts_offset = timestamp akumulatif
   322→   322→        cmd = [
   323→   323→            ffmpeg_path, '-y',
   324→   324→            '-hwaccel', 'videotoolbox',
   325→   325→            '-ss', f'{mp4_start:.3f}',
   326→   326→            '-i', mp4_path,
   327→   327→            '-t', f'{mp4_dur:.3f}',
   328→   328→            '-c:v', 'copy',
   329→   329→            '-an',
   330→   330→            '-output_ts_offset', f'{cumulative_offset:.6f}',
   331→   331→            '-fflags', '+genpts',
   332→   332→            output_file,
   333→   333→        ]
   334→   334→    else:
   335→   335→        # Re-encode: setpts untuk slow-mo + output_ts_offset untuk timestamp akumulatif
   336→   336→        # fps=30 filter SEBELUM setpts (VFR guard, dari riset VoiceStudio)
   337→   337→        #
   338→   338→        # Catatan test #16 (5 Okt 2026): tpad=stop_mode=clone untuk cap slow-mo 2x
   339→   339→        # GAGAL — PTS overflow (DTS 141670994490979984, INT64_MAX/1000) di h264_videotoolbox.
   340→   340→        # Cloned frames dapet PTS yang overflow saat di-mux → Pass 2 drop frames invalid.
   341→   341→        # 16 cue dengan factor > 2x sekarang frame hilang (lebih buruk dari stop-motion).
   342→   342→        # Keputusan: REVERT tpad. Slow-mo factor > 2x kembali ke setpts * factor (stop-motion,
   343→   343→        # tapi frame ada). Skip threshold 100B tetap (fix #16 yang bagus).
   344→   344→        # Future: kalau mau smooth slow-mo > 2x, pakai minterpolate (CPU heavy).
   345→   345→        cmd = [
   346→   346→            ffmpeg_path, '-y',
   347→   347→            '-hwaccel', 'videotoolbox',
   348→   348→            '-i', mp4_path,
   349→   349→            '-ss', f'{mp4_start:.3f}',
   350→   350→            '-t', f'{target_dur:.3f}',
   351→   351→            '-vf', f'fps=30,setpts=(PTS-STARTPTS)*{factor:.6f}',
   352→   352→            *enc_params,
   353→   353→            '-output_ts_offset', f'{cumulative_offset:.6f}',
   354→   354→            '-an',
   355→   355→            '-fflags', '+genpts',
   356→   356→            output_file,
   357→   357→        ]
   358→   358→
   359→   359→    try:
   360→   360→        result = subprocess.run(cmd, capture_output=True, text=True, timeout=300)
   361→   361→        if result.returncode != 0:
   362→   362→            # Print stderr LENGKAP untuk debugging (bukan cuma [-300:])
   363→   363→            # Cari baris yang mengandung "Error" atau "error" di stderr
   364→   364→            stderr_lines = result.stderr.split('\n')
   365→   365→            error_lines = [l for l in stderr_lines if 'error' in l.lower() or 'invalid' in l.lower() or 'not found' in l.lower() or 'no such' in l.lower()]
   366→   366→            error_summary = error_lines[0] if error_lines else stderr_lines[-5:]
   367→   367→            return {
   368→   368→                'index': seg_idx,
   369→   369→                'output_file': output_file,
   370→   370→                'status': 'failed',
   371→   371→                'error': f'code={result.returncode} | {error_summary[:200]}',
   372→   372→            }
   373→   373→        # VALIDASI: cek duration output — hanya untuk re-encode segments
   374→   374→        # (stream copy durasi = source durasi, tidak perlu validasi terhadap target)
   375→   375→        # SKIP validasi untuk sekarang. Segment terakhir sering terpotong
   376→   376→        # karena mp4_end > mp4_duration (video input lebih pendek dari target).
   377→   377→        # Concat di Pass 2 akan handle — segment pendek tetap di-concat.
   378→   378→        return {'index': seg_idx, 'output_file': output_file, 'status': 'ok'}
   379→   379→    except subprocess.TimeoutExpired:
   380→   380→        return {
   381→   381→            'index': seg_idx,
   382→   382→            'output_file': output_file,
   383→   383→            'status': 'failed',
   384→   384→            'error': 'timeout (300s)',
   385→   385→        }
   386→   386→    except Exception as e:
   387→   387→        return {
   388→   388→            'index': seg_idx,
   389→   389→            'output_file': output_file,
   390→   390→            'status': 'failed',
   391→   391→            'error': str(e),
   392→   392→        }
   393→   393→

   395→   395→# ============================================================
   396→   396→# Pass 1: Render semua segments (parallel)

   399→   399→def render_all_segments(tasks, mp4_path, segments_dir, ffmpeg_path, ffprobe_path, preset, workers=4, encoder='libx264'):
   400→   400→    """
   401→   401→    Render semua segments secara parallel.
   402→   402→    Pakai ProcessPoolExecutor untuk parallel FFmpeg processes.
   403→   403→    """
   404→   404→    total = len(tasks)
   405→   405→    print(f'\n=== Pass 1: Render {total} segments (parallel {workers} workers) ===')
   406→   406→
   407→   407→    completed = 0
   408→   408→    failed = 0
   409→   409→    start_time = time.time()
   410→   410→
   411→   411→    with ProcessPoolExecutor(max_workers=workers) as executor:
   412→   412→        # Submit semua tasks
   413→   413→        futures = {
   414→   414→            executor.submit(render_segment, task, mp4_path, segments_dir, ffmpeg_path, ffprobe_path, preset, True, encoder): task
   415→   415→            for task in tasks
   416→   416→        }
   417→   417→
   418→   418→        # Wait dan report progress
   419→   419→        for future in as_completed(futures):
   420→   420→            task = futures[future]
   421→   421→            try:
   422→   422→                result = future.result()
   423→   423→                completed += 1
   424→   424→                if result['status'] == 'failed':
   425→   425→                    failed += 1
   426→   426→                    print(f'  ✗ Segment {result["index"]} gagal: {result.get("error", "unknown")[:100]}')
   427→   427→                elif result['status'] == 'skipped':
   428→   428→                    pass  # silent skip
   429→   429→
   430→   430→                # Progress setiap 50 segments
   431→   431→                if completed % 50 == 0 or completed == total:
   432→   432→                    elapsed = time.time() - start_time
   433→   433→                    rate = completed / elapsed if elapsed > 0 else 0
   434→   434→                    eta = (total - completed) / rate if rate > 0 else 0
   435→   435→                    print(f'  → {completed}/{total} segments ({100*completed/total:.1f}%) — '
   436→   436→                          f'elapsed {elapsed:.0f}s, ETA {eta:.0f}s, failed {failed}')
   437→   437→            except Exception as e:
   438→   438→                failed += 1
   439→   439→                print(f'  ✗ Task exception: {e}')
   440→   440→
   441→   441→    print(f'\n=== Pass 1 selesai ===')
   442→   442→    print(f'  Total: {total} segments')
   443→   443→    print(f'  Sukses: {total - failed}')
   444→   444→    print(f'  Gagal: {failed}')
   445→   445→    print(f'  Waktu: {time.time() - start_time:.0f}s')
   446→   446→
   447→   447→    # Allow up to 5% failure (segment pendek di akhir video wajar gagal)
   448→   448→    # Kalau gagal > 5%, return False (gagal total)
   449→   449→    # Kalau gagal <= 5%, return True (lanjut ke Pass 2, segment gagal di-skip di concat)
   450→   450→    if failed > total * 0.05:
   451→   451→        print(f'  ❌ Gagal {failed}/{total} ({100*failed/total:.1f}%) > 5% threshold')
   452→   452→        return False
   453→   453→    elif failed > 0:
   454→   454→        print(f'  ⚠ {failed} segment gagal ({100*failed/total:.1f}%), tapi < 5% → lanjut ke Pass 2 (segment gagal di-skip)')
   455→   455→    return True

   462→   462→def concat_segments(tasks, segments_dir, audio_jawa, output, ffmpeg_path, has_audio_ori, ducking_db, video_only=False, encoder='libx264'):
   463→   463→    """
   464→   464→    Concat semua segment files menggunakan concat demuxer.
   465→   465→    Kalau video_only=True: output = video tanpa audio (user import audio terpisah di DaVinci).
   466→   466→    """
   467→   467→    print(f'\n=== Pass 2: Concat segments + mix audio ===')
   468→   468→
   469→   469→    # Build concat list file — skip segments yang gagal (file tidak ada atau terlalu kecil)
   470→   470→    # FIX test #15: threshold 1000 bytes terlalu tinggi, segment pendek (gap 0.05s, cue 0.17s)
   471→   471→    # bisa hasilkan file < 1KB → di-skip → gap di video → "berhenti lama" di VLC/DaVinci.
   472→   472→    # Lower ke 100 bytes supaya segment pendek tetap masuk concat.
   473→   473→    MIN_SEGMENT_SIZE = 100  # bytes (sebelumnya 1000)
   474→   474→    concat_list = os.path.join(segments_dir, 'concat_list.txt')
   475→   475→    included = 0
   476→   476→    skipped = 0
   477→   477→    with open(concat_list, 'w') as f:
   478→   478→        for task in tasks:
   479→   479→            seg_file = os.path.join(segments_dir, f'seg_{task["index"]:05d}.mp4')
   480→   480→            # Skip kalau file tidak ada atau terlalu kecil (segment gagal total)
   481→   481→            if os.path.isfile(seg_file) and os.path.getsize(seg_file) > MIN_SEGMENT_SIZE:
   482→   482→                f.write(f"file '{seg_file}'\n")
   483→   483→                included += 1
   484→   484→            else:
   485→   485→                skipped += 1
   486→   486→
   487→   487→    print(f'  → Concat list: {concat_list} ({included} files, {skipped} skipped)')
   488→   488→
   489→   489→    # Encoder params untuk Pass 2 re-encode
   490→   490→    if encoder == 'h264_videotoolbox':
   491→   491→        enc_params = ['-c:v', 'h264_videotoolbox', '-b:v', '5M', '-realtime', '0',
   492→   492→                       '-bf', '0', '-profile:v', 'high']
   493→   493→    else:
   494→   494→        enc_params = ['-c:v', 'libx264', '-preset', 'fast', '-crf', '23',
   495→   495→                       '-bf', '0', '-profile:v', 'high']
   496→   496→
   497→   497→    # Pass 2: STREAM COPY (instant, 2 detik)
   498→   498→    # Timestamp sudah akumulatif dari Pass 1 (output_ts_offset)
   499→   499→    # → concat demuxer stream copy = seamless, tidak ada DTS out of order
   500→   500→    if video_only:
   501→   501→        print(f'  → Mode: VIDEO ONLY (stream copy, tanpa audio)')
   502→   502→        cmd = [
   503→   503→            ffmpeg_path, '-y',
   504→   504→            '-f', 'concat', '-safe', '0',
   505→   505→            '-i', concat_list,
   506→   506→            '-c:v', 'copy',
   507→   507→            '-an',
   508→   508→            '-movflags', '+faststart',
   509→   509→            '-timecode', '00:00:00:00',
   510→   510→            output,
   511→   511→        ]
   512→   512→    else:
   513→   513→        cmd = [
   514→   514→            ffmpeg_path, '-y',
   515→   515→            '-f', 'concat', '-safe', '0',
   516→   516→            '-i', concat_list,
   517→   517→            '-i', audio_jawa,
   518→   518→            '-map', '0:v',
   519→   519→            '-map', '1:a',
   520→   520→            '-c:v', 'copy',
   521→   521→            '-c:a', 'aac',
   522→   522→            '-b:a', '192k',
   523→   523→            '-movflags', '+faststart',
   524→   524→            '-timecode', '00:00:00:00',
   525→   525→            '-shortest',
   526→   526→            output,
   527→   527→        ]
   528→   528→
   529→   529→    print(f'  → Concat + mix audio...')
   530→   530→    print(f'  Command: {" ".join(cmd[:5])} ...')
   531→   531→
   532→   532→    result = subprocess.run(cmd)
   533→   533→    if result.returncode != 0:
   534→   534→        print(f'\n✗ Pass 2 gagal dengan code {result.returncode}')
   535→   535→        return False
   536→   536→
   537→   537→    print(f'  ✓ Pass 2 selesai')
   538→   538→    return True
