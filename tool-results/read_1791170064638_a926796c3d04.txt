   103→   103→/**
   104→   104→ * Synthesize text. Edge TTS: natural rate (rate='+0%').
   105→   105→ * Speed up/slow down di-handle client-side dengan OfflineAudioContext + preservePitch.

   107→   107→export async function synthesizeText(
   108→   108→  text: string,
   109→   109→  opts: {
   110→   110→    provider: Provider
   111→   111→    voice: string
   112→   112→    model?: string
   113→   113→    apiKey?: string
   114→   114→    speed?: number           // Kokoro speed (default 1.0)
   115→   115→    rate?: string             // Edge TTS prosody rate (default '+0%' = natural)
   116→   116→    openaiSpeed?: number      // OpenAI/OpenRouter speed (0.25-4.0)
   117→   117→    onModelProgress?: (p: TTSProgress) => void
   118→   118→  },
   119→   119→): Promise<{ audioBlob: Blob; mimeType: string; pcm?: Float32Array; sampleRate?: number }> {
   120→   120→  if (!text.trim()) {
   121→   121→    return { audioBlob: new Blob([]), mimeType: 'audio/mp3' }
   122→   122→  }
   123→   123→
   124→   124→  switch (opts.provider) {
   125→   125→    case 'edge':
   126→   126→      // Edge TTS: natural rate, speed up di-handle client-side
   127→   127→      return {
   128→   128→        audioBlob: await edgeTTS(text, opts.voice, { rate: opts.rate || '+0%' }),
   129→   129→        mimeType: 'audio/mp3',
   130→   130→      }
   131→   131→    case 'kokoro': {
   132→   132→      await ensureKokoroModel(opts.onModelProgress)
   133→   133→      const speed = opts.speed || 1.0
   134→   134→      const result = await kokoroSynth(text, opts.voice || DEFAULT_KOKORO_VOICE, speed)
   135→   135→      return { audioBlob: new Blob([]), mimeType: 'audio/wav', pcm: result.audio, sampleRate: result.sampleRate }
   136→   136→    }
   137→   137→    case 'openai':
   138→   138→      if (!opts.apiKey) throw new Error('OpenAI API key belum diisi')
   139→   139→      return {
   140→   140→        audioBlob: await openaiTTS(text, opts.voice, opts.apiKey!, 'tts-1-hd', opts.openaiSpeed || 1.0),
   141→   141→        mimeType: 'audio/mp3',
   142→   142→      }
   143→   143→    case 'openrouter':
   144→   144→      if (!opts.apiKey) throw new Error('OpenRouter API key belum diisi')
   145→   145→      return {
   146→   146→        audioBlob: await openRouterTTS(text, opts.model || 'openai/tts-1-hd', opts.voice, opts.apiKey!, opts.openaiSpeed || 1.0),
   147→   147→        mimeType: 'audio/mp3',
   148→   148→      }
   149→   149→    default:
   150→   150→      throw new Error(`Unknown provider: ${opts.provider}`)
   151→   151→  }
   152→   152→}

   154→   154→/**
   155→   155→ * Decode synth result + convert ke mono + resample + TRIM SILENCE.
   156→   156→ *
   157→   157→ * Port dari ThioJoe trim_clip (audio_builder.py):
   158→   158→ * - detect_leading_silence → trim awal
   159→   159→ * - reverse → detect_leading_silence → trim akhir
   160→   160→ * - + 50ms padding di awal/akhir supaya tidak abrupt
   161→   161→ *
   162→   162→ * Returns Float32Array (resampled ke OUTPUT_SAMPLE_RATE, trimmed).
   163→   163→ */
   164→   164→async function decodeMonoTrimResample(
   165→   165→  synth: { audioBlob: Blob; mimeType: string; pcm?: Float32Array; sampleRate?: number },
   166→   166→  sampleRate: number = OUTPUT_SAMPLE_RATE,
   167→   167→  trimSilenceEnabled: boolean = true,
   168→   168→): Promise<{ audio: Float32Array; durationSec: number; rawDurationSec: number; trimmedDurationSec: number }> {
   169→   169→  let audioBuffer: AudioBuffer
   170→   170→  if (synth.pcm) {
   171→   171→    if (synth.pcm.length === 0) throw new Error('Empty audio output')
   172→   172→    const pcmSampleRate = synth.sampleRate || sampleRate
   173→   173→    const ctx = new AudioContext({ sampleRate: pcmSampleRate })
   174→   174→    audioBuffer = ctx.createBuffer(1, synth.pcm.length, pcmSampleRate)
   175→   175→    audioBuffer.copyToChannel(synth.pcm, 0)
   176→   176→    ctx.close()
   177→   177→  } else {
   178→   178→    if (synth.audioBlob.size === 0) throw new Error('Empty audio output')
   179→   179→    audioBuffer = await decodeAudioBlob(synth.audioBlob, sampleRate)
   180→   180→  }
   181→   181→
   182→   182→  const rawDurationSec = audioBuffer.duration
   183→   183→  let mono = toMono(audioBuffer)
   184→   184→  if (audioBuffer.sampleRate !== sampleRate) {
   185→   185→    mono = linearResample(mono, audioBuffer.sampleRate, sampleRate)
   186→   186→  }
   187→   187→
   188→   188→  let trimmed: Float32Array
   189→   189→  if (trimSilenceEnabled && mono.length > 0) {
   190→   190→    trimmed = trimSilence(mono, -30, sampleRate, 50)
   191→   191→  } else {
   192→   192→    trimmed = mono
   193→   193→  }
   194→   194→  const trimmedDurationSec = trimmed.length / sampleRate
   195→   195→
   196→   196→  return {
   197→   197→    audio: trimmed,
   198→   198→    durationSec: trimmedDurationSec,
   199→   199→    rawDurationSec,
   200→   200→    trimmedDurationSec,
   201→   201→  }
   202→   202→}
   203→   203→
   204→   204→/**
   205→   205→ * Speed up atau slow down TIDAK di client — OfflineAudioContext + preservePitch
   206→   206→ * TIDAK berfungsi di Brave. Return audio asli.
   207→   207→ *
   208→   208→ * Kita pakai TWO-PASS Edge TTS server-side rate instead.
   209→   209→ */
   210→   210→async function timeStretchClient(
   211→   211→  audioBuffer: AudioBuffer,
   212→   212→  _ratio: number,
   213→   213→  sampleRate: number,
   214→   214→): Promise<Float32Array> {
   215→   215→  // TIDAK speed up di client — preservePitch tidak berfungsi di OfflineAudioContext Brave
   216→   216→  // Return audio asli (natural). Speed up di-handle via TWO-PASS Edge TTS.
   217→   217→  const mono = toMono(audioBuffer)
   218→   218→  if (audioBuffer.sampleRate !== sampleRate) {
   219→   219→    return linearResample(mono, audioBuffer.sampleRate, sampleRate)
   220→   220→  }
   221→   221→  return mono
   222→   222→}
   223→   223→
   224→   224→/**
   225→   225→ * Format ratio jadi Edge TTS rate percent string.
   226→   226→ * ratio 1.0 = +0%, 1.5 = +50%, 0.8 = -20%
   227→   227→ */
   228→   228→function formatEdgeRate(ratio: number): string {
   229→   229→  // Clamp ke range yang Edge TTS support: -50% sampai +200% (ratio 0.5-3.0)
   230→   230→  const clamped = Math.min(Math.max(ratio, 0.5), 3.0)
   231→   231→  const percent = Math.round((clamped - 1) * 100)
   232→   232→  return (percent >= 0 ? '+' : '') + percent + '%'
   233→   233→}
   234→   234→
   235→   235→/**
   236→   236→ * Generate narration audio for SRT entries, stitched into one WAV.
   237→   237→ *
   238→   238→ * DUA MODE (filosofi Voicertool.com/subs):
   239→   239→ *
   240→   240→ * ON (respectTiming=true): audio fit ke cue duration, durasi = SRT
   241→   241→ * - TWO-PASS Edge TTS: generate natural → ukur → re-generate dengan rate kalau perlu
   242→   242→ * - Speed mode (opts.speedMode):
   243→   243→ *   • "speedup-only": speed up kalau audio lebih panjang dari cue,
   244→   244→ *     biarkan silence di akhir cue kalau lebih pendek (natural hening).
   245→   245→ *   • "speedup-slowdown" (DEFAULT — filosofi Voicertool kombinasi):
   246→   246→ *     speed up kalau lebih panjang, SLOW DOWN kalau lebih pendek —
   247→   247→ *     cue selalu terisi penuh, tidak ada silence.
   248→   248→ * - Clamp slowdown: minimum 0.7x (Edge rate -30%) supaya audio tidak kedengaran aneh.
   249→   249→ * - Audio DIPERTAHANKAN UTUH — TIDAK ADA TRUNCATION
   250→   250→ * - Kalau audio cue lebih panjang dari cue → tumpuk dengan CROSSFADE:
   251→   251→ *   cue N fade out di region overlap, cue N+1 fade in di region overlap
   252→   252→ * - Position = entry.start (SRT timing WAJIB)
   253→   253→ * - Total durasi = SRT end time
   254→   254→ *
   255→   255→ * OFF (respectTiming=false): audio natural alami (TANPA POTONGAN)
   256→   256→ * - Sequential playback, audio utuh 100%
   257→   257→ * - Total durasi = sum semua audio
   258→   258→ *
   259→   259→ * Output: WAV 24kHz mono 16-bit PCM.
   260→   260→ */
   261→   261→export async function narrateEntries(
   262→   262→  entries: SrtEntry[],
   263→   263→  opts: NarrationOptions,
   264→   264→): Promise<NarrationResult> {
   265→   265→  if (entries.length === 0) throw new Error('No subtitles to narrate')
   266→   266→
   267→   267→  opts.onStage?.({ stage: 'synthesizing', message: 'Mulai synthesizing…', percent: 0 })
   268→   268→  const total = entries.length
   269→   269→  // Tiap segment: { position, audio, cueStart, cueEnd }
   270→   270→  const placedSegments: { position: number; audio: Float32Array; cueStart: number; cueEnd: number }[] = []
   271→   271→  let cursor = 0
   272→   272→  let successCount = 0
   273→   273→  let failCount = 0
   274→   274→  let firstError = ''
   275→   275→
   276→   276→  for (let i = 0; i < entries.length; i++) {
   277→   277→    const entry = entries[i]
   278→   278→    const text = entry.textLines.join(' ').trim()
   279→   279→
   280→   280→    if (!text) {
   281→   281→      if (opts.respectTiming) {
   282→   282→        cursor = Math.max(cursor, Math.floor(entry.end * OUTPUT_SAMPLE_RATE))
   283→   283→      } else {
   284→   284→        cursor += Math.floor(0.3 * OUTPUT_SAMPLE_RATE)
   285→   285→      }
   286→   286→      opts.onLineProgress?.(i + 1, total, '(empty)')
   287→   287→      continue
   288→   288→    }
   289→   289→
   290→   290→    opts.onLineProgress?.(i + 1, total, text.slice(0, 60))
   291→   291→    opts.onStage?.({
   292→   292→      stage: 'synthesizing',
   293→   293→      message: `Baris ${i + 1}/${total}: "${text.slice(0, 40)}${text.length > 40 ? '…' : ''}"`,
   294→   294→      percent: (i / total) * 100,
   295→   295→    })
   296→   296→
   297→   297→    try {
   298→   298→      let finalAudio: Float32Array
   299→   299→      let position: number
   300→   300→
   301→   301→      if (opts.respectTiming) {
   302→   302→        // === ON MODE: TRIM SILENCE → NATURAL-FIRST + speed up only kalau tabrakan ===
   303→   303→        // Port dari ThioJoe audio_builder.py:
   304→   304→        // 1. Generate TTS natural
   305→   305→        // 2. TRIM SILENCE di awal/akhir (detect_leading_silence -30dB + 50ms padding)
   306→   306→        // 3. Hitung ratio berdasarkan TRIMMED duration (bukan raw TTS output)
   307→   307→        // 4. Kalau trimmed ≤ cue → pakai natural (tanpa speed up, tanpa silence buatan)
   308→   308→        // 5. Kalau trimmed > cue → speed up via Edge TTS server-side rate
   309→   309→        const cueDuration = entry.end - entry.start
   310→   310→        const cueStartSamples = Math.floor(entry.start * OUTPUT_SAMPLE_RATE)
   311→   311→        const cueEndSamples = Math.floor(entry.end * OUTPUT_SAMPLE_RATE)
   312→   312→
   313→   313→        // Hitung ruang yang tersedia: dari entry.start sampai cue berikutnya mulai
   314→   314→        // Kalau ini cue terakhir, ruang = cue end (SRT duration)
   315→   315→        const nextEntryStart = (i + 1 < entries.length) ? entries[i + 1].start : entry.end
   316→   316→        const availableDuration = nextEntryStart - entry.start
   317→   317→
   318→   318→        // PASS 1: Generate natural audio untuk ukur actual duration
   319→   319→        const synth1 = await synthesizeText(text, { ...opts, rate: '+0%', openaiSpeed: 1.0, speed: 1.0 })
   320→   320→
   321→   321→        // Decode + mono + resample + TRIM SILENCE
   322→   322→        const pass1 = await decodeMonoTrimResample(synth1, OUTPUT_SAMPLE_RATE, true)
   323→   323→        const actualDuration = pass1.durationSec // TRIMMED duration (bukan raw TTS)
   324→   324→
   325→   325→        // SKEMA 1: NATURAL-FIRST (untuk speedup-only)
   326→   326→        // Mode "speedup-only" (seperti Voicertool): biarkan silence di akhir cue kalau audio lebih pendek.
   327→   327→        // Mode "speedup-slowdown" (seperti Voicertool): kalau audio lebih pendek dari cue, SLOW DOWN supaya fit penuh.
   328→   328→        const speedMode = opts.speedMode || 'speedup-slowdown' // default = filosofi Voicertool kombinasi
   329→   329→        const MIN_SLOWDOWN_RATIO = 0.7 // batas bawah: audio tidak boleh lebih lambat dari 0.7x (kedengaran aneh kalau terlalu lambat)
   330→   330→
   331→   331→        if (actualDuration <= cueDuration) {
   332→   332→          // Audio muat di cue (entry.start → entry.end)
   333→   333→          if (speedMode === 'speedup-slowdown' && actualDuration < cueDuration * 0.95) {
   334→   334→            // Mode Voicertool "speed up and slow down": SLOW DOWN untuk isi cue penuh (hilangkan silence)
   335→   335→            // Ratio: actualDuration / cueDuration (e.g., 0.7 → Edge rate = -30%)
   336→   336→            // Clamp ke MIN_SLOWDOWN_RATIO supaya tidak terlalu lambat
   337→   337→            const rawRatio = actualDuration / cueDuration
   338→   338→            const slowRatio = Math.max(rawRatio, MIN_SLOWDOWN_RATIO)
   339→   339→            // Edge TTS rate: ratio 0.7 → -30%, ratio 1.0 → +0%
   340→   340→            const edgeRate = formatEdgeRate(slowRatio)
   341→   341→
   342→   342→            if (opts.provider === 'edge' && slowRatio < 0.97) {
   343→   343→              // Re-generate dengan Edge TTS server-side rate (pitch dipertahankan di server)
   344→   344→              const synthSlow = await synthesizeText(text, {
   345→   345→                ...opts,
   346→   346→                rate: edgeRate,
   347→   347→                openaiSpeed: slowRatio,
   348→   348→                speed: slowRatio,
   349→   349→              })
   350→   350→              // Pass 2 juga di-trim supaya konsisten
   351→   351→              const passSlow = await decodeMonoTrimResample(synthSlow, OUTPUT_SAMPLE_RATE, true)
   352→   352→              finalAudio = passSlow.audio
   353→   353→            } else {
   354→   354→              // Provider non-Edge atau ratio sudah dekat 1.0 → pakai natural (trimmed pass 1)
   355→   355→              finalAudio = pass1.audio
   356→   356→            }
   357→   357→          } else {
   358→   358→            // speedup-only: pakai natural yang sudah di-trim (hening asli TTS sudah hilang,
   359→   359→            // tapi kalau trimmed masih lebih pendek dari cue, sisanya tetap hening — itu natural SRT)
   360→   360→            finalAudio = pass1.audio
   361→   361→          }
   362→   362→        }
   363→   363→        // SKEMA 2: SPEED UP (kalau audio TIDAK muat di cue)
   364→   364→        else if (opts.provider === 'edge') {
   365→   365→          // Hitung ratio berdasarkan ruang tersedia, BUKAN cue duration saja.
   366→   366→          // Ruang = dari entry.start sampai cue berikutnya mulai (atau cue.end kalau terakhir).
   367→   367→          // Kalau ruang = 7s, audio = 10s → ratio = 10/7 = 1.43x
   368→   368→          // Kalau ruang = cue = 5s, audio = 10s → ratio = 10/5 = 2.0x
   369→   369→          const targetDuration = Math.max(cueDuration, availableDuration)
   370→   370→          const ratio = actualDuration / targetDuration
   371→   371→          const edgeRate = formatEdgeRate(ratio)
   372→   372→
   373→   373→          // Kalau ratio kecil (audio sedikit lebih panjang dari ruang), pakai natural + crossfade
   374→   374→          if (ratio <= 1.1) {
   375→   375→            // Audio hanya sedikit lebih panjang — pakai natural (trimmed), crossfade handle
   376→   376→            finalAudio = pass1.audio
   377→   377→          } else {
   378→   378→            // Speed up needed — re-generate dengan Edge TTS server-side rate (pitch preserved)
   379→   379→            const synth2 = await synthesizeText(text, {
   380→   380→              ...opts,
   381→   381→              rate: edgeRate,
   382→   382→              openaiSpeed: ratio,
   383→   383→              speed: ratio,
   384→   384→            })
   385→   385→            // Pass 2 juga di-trim
   386→   386→            const pass2 = await decodeMonoTrimResample(synth2, OUTPUT_SAMPLE_RATE, true)
   387→   387→            finalAudio = pass2.audio
   388→   388→          }
   389→   389→        } else {
   390→   390→          // OpenAI/OpenRouter/Kokoro: pakai audio dari pass 1 (sudah natural + trimmed)
   391→   391→          finalAudio = pass1.audio
   392→   392→        }
   393→   393→
   394→   394→        position = cueStartSamples
   395→   395→        cursor = position + finalAudio.length
   396→   396→      } else {
   397→   397→        // === OFF MODE: natural audio dengan optional speed multiplier (trim silence juga) ===
   398→   398→        const offSpeed = opts.offSpeed || 1.0
   399→   399→        let offRate = '+0%'
   400→   400→        let offOpenaiSpeed = 1.0
   401→   401→        let offKokoroSpeed = 1.0
   402→   402→        if (offSpeed !== 1.0) {
   403→   403→          offRate = formatEdgeRate(offSpeed)
   404→   404→          offOpenaiSpeed = offSpeed
   405→   405→          offKokoroSpeed = offSpeed
   406→   406→        }
   407→   407→
   408→   408→        const synth = await synthesizeText(text, {
   409→   409→          ...opts,
   410→   410→          rate: offRate,
   411→   411→          openaiSpeed: offOpenaiSpeed,
   412→   412→          speed: offKokoroSpeed,
   413→   413→        })
   414→   414→
   415→   415→        // OFF mode juga trim silence supaya audio lebih rapat antar cue (tidak ada hening buatan di awal/akhir)
   416→   416→        const off = await decodeMonoTrimResample(synth, OUTPUT_SAMPLE_RATE, true)
   417→   417→        finalAudio = off.audio
   418→   418→        position = cursor
   419→   419→        cursor = position + finalAudio.length
   420→   420→      }
   421→   421→
   422→   422→      placedSegments.push({
   423→   423→        position,
   424→   424→        audio: finalAudio,
   425→   425→        cueStart: Math.floor(entry.start * OUTPUT_SAMPLE_RATE),
   426→   426→        cueEnd: Math.floor(entry.end * OUTPUT_SAMPLE_RATE),
   427→   427→      })
   428→   428→      successCount++
   429→   429→    } catch (e) {
   430→   430→      failCount++
   431→   431→      if (!firstError) firstError = (e as Error).message
   432→   432→      console.error('TTS failed for line', i, e)
   433→   433→    }
   434→   434→  }
   435→   435→
   436→   436→  if (successCount === 0) {
   437→   437→    throw new Error(
   438→   438→      `TTS gagal untuk semua ${total} baris. Error: ${firstError || 'unknown'}. ` +
   439→   439→      (opts.provider === 'edge'
   440→   440→        ? 'Coba ganti ke provider OpenAI atau OpenRouter.'
   441→   441→        : 'Cek API key atau koneksi internet.'),
   442→   442→    )
   443→   443→  }
   444→   444→
   445→   445→  if (failCount > 0) {
   446→   446→    console.warn(`TTS: ${successCount}/${total} berhasil, ${failCount} gagal.`)
   447→   447→  }
   448→   448→
   449→   449→  opts.onStage?.({ stage: 'stitching', message: 'Menjahit audio + crossfade…', percent: 90 })
   450→   450→
   451→   451→  // Hitung total samples
   452→   452→  let totalSamples: number
   453→   453→  if (opts.respectTiming) {
   454→   454→    totalSamples = Math.floor(entries[entries.length - 1].end * OUTPUT_SAMPLE_RATE)
   455→   455→  } else {
   456→   456→    let maxEnd = 0
   457→   457→    for (const seg of placedSegments) {
   458→   458→      maxEnd = Math.max(maxEnd, seg.position + seg.audio.length)
   459→   459→    }
   460→   460→    totalSamples = maxEnd
   461→   461→  }
   462→   462→
   463→   463→  // Build output dengan CROSSFADE
   464→   464→  const allAudio = new Float32Array(totalSamples)
   465→   465→
   466→   466→  for (let i = 0; i < placedSegments.length; i++) {
   467→   467→    const seg = placedSegments[i]
   468→   468→
   469→   469→    // Kalau ON mode dan audio extends beyond cueEnd (overlap dengan cue berikutnya)
   470→   470→    if (opts.respectTiming && seg.cueEnd > 0) {
   471→   471→      // Cek apakah ada cue berikutnya yang overlap
   472→   472→      const nextSeg = placedSegments[i + 1]
   473→   473→      if (nextSeg) {
   474→   474→        const overlapStart = seg.cueEnd // cue end = next cue start
   475→   475→        const overlapEnd = Math.min(seg.position + seg.audio.length, nextSeg.position + 100) // overlap region
   476→   476→
   477→   477→        if (overlapEnd > overlapStart && seg.position + seg.audio.length > overlapStart) {
   478→   478→          // Ada overlap — apply fade out ke akhir cue ini
   479→   479→          const fadeStart = overlapStart - seg.position
   480→   480→          const fadeEnd = Math.min(seg.audio.length, overlapEnd - seg.position)
   481→   481→          if (fadeEnd > fadeStart) {
   482→   482→            applyFadeOut(seg.audio, fadeStart, fadeEnd)
   483→   483→          }
   484→   484→
   485→   485→          // Apply fade in ke awal cue berikutnya
   486→   486→          const fadeInStart = 0
   487→   487→          const fadeInEnd = Math.min(nextSeg.audio.length, fadeEnd - fadeStart)
   488→   488→          if (fadeInEnd > fadeInStart) {
   489→   489→            applyFadeIn(nextSeg.audio, fadeInStart, fadeInEnd)
   490→   490→          }
   491→   491→        }
   492→   492→      }
   493→   493→    }
   494→   494→
   495→   495→    // MIX audio ke buffer (ADD, bukan overwrite)
   496→   496→    mixAudioInto(allAudio, seg.audio, seg.position)
   497→   497→  }
   498→   498→
   499→   499→  const blob = encodeWav(allAudio, OUTPUT_SAMPLE_RATE)
   500→   500→  const durationSec = allAudio.length / OUTPUT_SAMPLE_RATE
   501→   501→  opts.onStage?.({ stage: 'done', message: 'Narration selesai', percent: 100 })
   502→   502→  return { blob, sampleRate: OUTPUT_SAMPLE_RATE, durationSec, previewUrl: URL.createObjectURL(blob) }
   503→   503→}

   505→   505→/**
   506→   506→ * Linear resample Float32Array dari sample rate asal ke target.
   507→   507→ */
   508→   508→function linearResample(audio: Float32Array, fromRate: number, toRate: number): Float32Array {
   509→   509→  if (fromRate === toRate) return audio
   510→   510→  const ratio = toRate / fromRate
   511→   511→  const targetLength = Math.floor(audio.length * ratio)
   512→   512→  const result = new Float32Array(targetLength)
   513→   513→  for (let i = 0; i < targetLength; i++) {
   514→   514→    const srcIdx = i / ratio
   515→   515→    const idx0 = Math.floor(srcIdx)
   516→   516→    const idx1 = Math.min(idx0 + 1, audio.length - 1)
   517→   517→    const frac = srcIdx - idx0
   518→   518→    result[i] = audio[idx0] * (1 - frac) + audio[idx1] * frac
   519→   519→  }
   520→   520→  return result
   521→   521→}

   527→   527→// ============================================================
   528→   528→// DUBBING MODE — SRT Jawa sebagai ground truth
   529→   529→// ============================================================
   530→   530→//
   531→   531→// Filosofi: Audio natural → SRT baru ngikut audio → MP4 ngikut SRT baru.
   532→   532→//
   533→   533→// Algoritma:
   534→   534→// 1. Generate audio natural per cue (Edge TTS rate 1.0x/1.25x/1.5x sesuai pilihan)
   535→   535→// 2. Bangun SRT baru: cue.start = original.start + accumulated_offset
   536→   536→//    cue.end = cue.start + audio_duration (audio utuh, tidak dipotong, tidak ditambah silence)
   537→   537→// 3. Kalau audio overflow ke cue asli berikutnya → offset bertambah
   538→   538→// 4. Kalau ada gap cukup (zona pemandangan) → offset tidak berubah (natural)
   539→   539→// 5. Min gap antar cue (default 150ms) — kalau audio cue[n] terlalu deket dengan cue[n+1].start,
   540→   540→//    push back cue[n+1] supaya ada jeda natural
   541→   541→//
   542→   542→// Output: 3 file
   543→   543→// - audio-jawa.wav (audio utuh natural, 24kHz mono)
   544→   544→// - subs-jawa-new.srt (SRT BARU dengan timing dari audio)
   545→   545→// - retime-map.json (peta retim video untuk ffmpeg)
   546→   546→
   547→   547→export interface DubbingOptions {
   548→   548→  provider: Provider
   549→   549→  voice: string
   550→   550→  model?: string
   551→   551→  apiKey?: string
   552→   552→  speed?: number             // 1.0, 1.25, 1.5 (default 1.25x — bantu slow-mo video)
   553→   553→  minGapSec?: number         // default 0.15 (150ms)
   554→   554→  concurrency?: number       // default 3 — parallel generate untuk speed up
   555→   555→  onModelProgress?: (p: TTSProgress) => void
   556→   556→  onLineProgress?: (current: number, total: number, text: string) => void
   557→   557→  onStage?: ProgressCallback
   558→   558→}

   560→   560→export interface DubbingRetimePoint {
   561→   561→  cueIndex: number           // 0-based
   562→   562→  originalStart: number      // seconds
   563→   563→  originalEnd: number        // seconds
   564→   564→  newStart: number           // seconds (di SRT baru)
   565→   565→  newEnd: number             // seconds
   566→   566→  originalDuration: number   // seconds
   567→   567→  newDuration: number        // seconds
   568→   568→  factor: number             // newDuration / originalDuration (>1 = slow-mo, <1 = fast-forward)
   569→   569→  type: 'cue' | 'gap'        // cue = dialog, gap = jeda antar cue
   570→   570→  text?: string              // untuk cue
   571→   571→}

   577→   577→  retimeMap: {
   578→   578→    version: string
   579→   579→    source: string           // info SRT asli
   580→   580→    speed: number
   581→   581→    minGapSec: number
   582→   582→    totalOffsetSec: number
   583→   583→    originalDurationSec: number

   591→   591→/**
   592→   592→ * Generate audio Jawa natural + bangun SRT baru dengan timing ngikut audio.
   593→   593→ * Audio utuh 100% (tidak dipotong, tidak ditambah silence).
   594→   594→ */
   595→   595→export async function narrateDubbingMode(
   596→   596→  entries: SrtEntry[],
   597→   597→  opts: DubbingOptions,
   598→   598→): Promise<DubbingResult> {
   599→   599→  if (entries.length === 0) throw new Error('No subtitles to narrate')
   600→   600→
   601→   601→  // Default speed = 1.0 (NATURAL). User bilang dari awal: mau natural, video slow-mo OK.
   602→   602→  // Speed 1.25/1.5 hanya kalau user pilih manual di UI untuk kurangi slow-mo video.
   603→   603→  const speed = opts.speed ?? 1.0
   604→   604→  const minGapSec = opts.minGapSec ?? 0.15
   605→   605→  const total = entries.length
   606→   606→  const concurrency = opts.concurrency ?? 3 // parallel generate untuk speed up
   607→   607→
   608→   608→  // Edge TTS rate: 1.0x = +0%, 1.25x = +25%, 1.5x = +50%
   609→   609→  const edgeRate = formatEdgeRate(speed)
   610→   610→
   611→   611→  opts.onStage?.({ stage: 'synthesizing', message: 'Mulai Dubbing Mode…', percent: 0 })
   612→   612→
   613→   613→  // Simpan audio per cue + durasi aktual (pre-allocate untuk parallel fill)
   614→   614→  const cueAudios: { pcm: Float32Array; durationSec: number; text: string }[] = new Array(entries.length)
   615→   615→  for (let i = 0; i < entries.length; i++) {
   616→   616→    cueAudios[i] = { pcm: new Float32Array(0), durationSec: 0, text: '' }
   617→   617→  }
   618→   618→  let successCount = 0
   619→   619→  let failCount = 0
   620→   620→  let firstError = ''
   621→   621→  let doneCount = 0
   622→   622→
   623→   623→  // Helper untuk generate 1 cue
   624→   624→  const generateOne = async (i: number) => {
   625→   625→    const entry = entries[i]
   626→   626→    const text = entry.textLines.join(' ').trim()
   627→   627→
   628→   628→    if (!text) {
   629→   629→      doneCount++
   630→   630→      opts.onLineProgress?.(doneCount, total, '(empty)')
   631→   631→      return
   632→   632→    }
   633→   633→
   634→   634→    try {
   635→   635→      const synth = await synthesizeText(text, {
   636→   636→        provider: opts.provider,
   637→   637→        voice: opts.voice,
   638→   638→        model: opts.model,
   639→   639→        apiKey: opts.apiKey,
   640→   640→        rate: opts.provider === 'edge' ? edgeRate : undefined,
   641→   641→        openaiSpeed: opts.provider !== 'edge' ? speed : undefined,
   642→   642→        speed: opts.provider === 'kokoro' ? speed : undefined,
   643→   643→        onModelProgress: opts.onModelProgress,
   644→   644→      })
   645→   645→
   646→   646→      // Decode + mono + resample + TRIM SILENCE
   647→   647→      const dub = await decodeMonoTrimResample(synth, OUTPUT_SAMPLE_RATE, true)
   648→   648→
   649→   649→      // VALIDATE durationSec — kalau NaN/Infinity/too big, anggap corrupt, skip
   650→   650→      // Edge case: Edge TTS proxy kadang return MP3 corrupt, audioBuffer.length bisa raksasa
   651→   651→      // → dub.durationSec jadi 3.11e+35 → stitching throw "Invalid typed array length"
   652→   652→      // Max cue audio 300 detik (5 menit) — wajar untuk cue Jawa natural di scene panjang
   653→   653→      const MAX_CUE_DURATION_SEC = 300
   654→   654→      if (!isFinite(dub.durationSec) || dub.durationSec <= 0 || dub.durationSec > MAX_CUE_DURATION_SEC) {
   655→   655→        console.warn(`[Dubbing] cue ${i} skip — invalid durationSec: ${dub.durationSec} (text: "${text.slice(0, 40)}")`)
   656→   656→        failCount++
   657→   657→        if (!firstError) firstError = `cue ${i} durationSec invalid: ${dub.durationSec}`
   658→   658→        // cueAudios[i] sudah pre-allocated dengan durationSec=0 → akan skip di stitching
   659→   659→        return
   660→   660→      }
   661→   661→
   662→   662→      cueAudios[i] = { pcm: dub.audio, durationSec: dub.durationSec, text }
   663→   663→      successCount++
   664→   664→    } catch (e) {
   665→   665→      failCount++
   666→   666→      if (!firstError) firstError = (e as Error).message
   667→   667→      cueAudios[i] = { pcm: new Float32Array(0), durationSec: 0, text }
   668→   668→      console.error('Dubbing TTS failed for line', i, e)
   669→   669→    } finally {
   670→   670→      doneCount++
   671→   671→      // Update progress per cue complete (lebih sering dari sebelumnya)
   672→   672→      opts.onLineProgress?.(doneCount, total, text.slice(0, 60))
   673→   673→      opts.onStage?.({
   674→   674→        stage: 'synthesizing',
   675→   675→        message: `Dubbing ${doneCount}/${total} cues — "${text.slice(0, 40)}${text.length > 40 ? '…' : ''}"`,
   676→   676→        percent: (doneCount / total) * 90, // max 90% (10% untuk stitching)
   677→   677→      })
   678→   678→    }
   679→   679→  }
   680→   680→
   681→   681→  // Process in batches of `concurrency` (parallel within batch, sequential across batches)
   682→   682→  // Edge TTS via Vercel proxy: 3 concurrent aman (Microsoft rate limit ~10 req/sec)
   683→   683→  for (let i = 0; i < entries.length; i += concurrency) {
   684→   684→    const batch = []
   685→   685→    for (let j = i; j < Math.min(i + concurrency, entries.length); j++) {
   686→   686→      batch.push(generateOne(j))
   687→   687→    }
   688→   688→    await Promise.all(batch)
   689→   689→  }
   690→   690→
   691→   691→  if (successCount === 0) {
   692→   692→    throw new Error(
   693→   693→      `TTS gagal untuk semua ${total} baris. Error: ${firstError || 'unknown'}. ` +
   694→   694→      (opts.provider === 'edge'
   695→   695→        ? 'Coba ganti ke provider OpenAI atau OpenRouter.'
   696→   696→        : 'Cek API key atau koneksi internet.'),
   697→   697→    )
   698→   698→  }
   699→   699→
   700→   700→  opts.onStage?.({ stage: 'stitching', message: 'Bangun SRT baru + stitch audio…', percent: 95 })
   701→   701→
   702→   702→  // === DEBUG: log cueAudios summary sebelum stitching ===
   703→   703→  // Untuk pin-point cue mana yang punya duration raksasa yang lolos dari Layer 1
   704→   704→  const cueAudiosSummary = cueAudios.map((c, i) => ({
   705→   705→    i,
   706→   706→    dur: c.durationSec,
   707→   707→    pcmLen: c.pcm.length,
   708→   708→    textLen: c.text.length,
   709→   709→  })).filter(c => !isFinite(c.dur) || c.dur > 60 || c.dur < 0)
   710→   710→  console.log('[Dubbing] Pre-stitching audit:', {
   711→   711→    totalCues: entries.length,
   712→   712→    successCount,
   713→   713→    failCount,
   714→   714→    suspiciousCues: cueAudiosSummary, // cue yang punya duration invalid
   715→   715→  })
   716→   716→  if (cueAudiosSummary.length > 0) {
   717→   717→    console.warn('[Dubbing] WARNING: ada cue dengan duration invalid yang akan di-skip di stitching')
   718→   718→  }
   719→   719→
   720→   720→  // === STITCHING: bangun SRT baru + mix audio ke buffer + encode WAV ===
   721→   721→  // Wrap dalam try-catch supaya error message spesifik ke user (tahap mana yang gagal)
   722→   722→  let indexedNewEntries: { start: number; end: number; text: string; cueIndex: number }[] = []
   723→   723→  let retimePoints: DubbingRetimePoint[] = []
   724→   724→  let offset = 0
   725→   725→  let blob: Blob
   726→   726→  let audioDurationSec: number
   727→   727→  let srtContent: string
   728→   728→  let retimeMap: { version: string; source: string; speed: number; minGapSec: number; totalOffsetSec: number; originalDurationSec: number; newDurationSec: number; points: DubbingRetimePoint[] }
   729→   729→
   730→   730→  try {
   731→   731→    // Step 1: Bangun timeline baru (cue index, start, end, retime points)
   732→   732→    console.log('[Dubbing] Stitching step 1: build timeline...', { totalCues: entries.length, successCount })
   733→   733→    for (let i = 0; i < entries.length; i++) {
   734→   734→      const entry = entries[i]
   735→   735→      const cueAudio = cueAudios[i]
   736→   736→      const text = entry.textLines.join(' ').trim()
   737→   737→
   738→   738→      // Skip cue kosong, atau duration invalid (NaN/Infinity/0/negatif)
   739→   739→      if (!text || !isFinite(cueAudio.durationSec) || cueAudio.durationSec <= 0) continue
   740→   740→
   741→   741→      // Sanity check: cue audio tidak boleh lebih dari 300 detik (5 menit)
   742→   742→      // Realistis max ~30s untuk cue panjang Jawa krama
   743→   743→      if (cueAudio.durationSec > 300) {
   744→   744→        console.warn(`[Dubbing] Stitching: cue ${i} durationSec ${cueAudio.durationSec}s > 300s, skip (text: "${text.slice(0, 40)}")`)
   745→   745→        continue
   746→   746→      }
   747→   747→
   748→   748→      // Cue baru: start = original.start + offset
   749→   749→      const newStart = entry.start + offset
   750→   750→      const newEnd = newStart + cueAudio.durationSec
   751→   751→      // Safety: kalau newEnd tidak finite, skip cue ini
   752→   752→      if (!isFinite(newEnd) || !isFinite(newStart)) {
   753→   753→        console.warn(`[Dubbing] Stitching: cue ${i} newStart/newEnd not finite: ${newStart}/${newEnd}, skip`)
   754→   754→        continue
   755→   755→      }
   756→   756→      // HARD CAP: kalau newEnd > 10 jam (36000s), skip cue + RESET offset ke 0
   757→   757→      // Ini mencegah offset accumulative dari cue corrupt sebelumnya
   758→   758→      // 10 jam = max wajar untuk SRT panjang (movie 3 jam + offset natural)
   759→   759→      if (newEnd > 36000) {
   760→   760→        console.warn(`[Dubbing] Stitching: cue ${i} newEnd ${newEnd}s > 36000s (10 jam), skip + reset offset from ${offset} to 0`)
   761→   761→        offset = 0
   762→   762→        continue
   763→   763→      }
   764→   764→      indexedNewEntries.push({ start: newStart, end: newEnd, text, cueIndex: i })
   765→   765→
   766→   766→      retimePoints.push({
   767→   767→        cueIndex: i,
   768→   768→        originalStart: entry.start,
   769→   769→        originalEnd: entry.end,
   770→   770→        newStart,
   771→   771→        newEnd,
   772→   772→        originalDuration: entry.end - entry.start,
   773→   773→        newDuration: cueAudio.durationSec,
   774→   774→        factor: cueAudio.durationSec / Math.max(0.001, entry.end - entry.start),
   775→   775→        type: 'cue',
   776→   776→        text,
   777→   777→      })
   778→   778→
   779→   779→      // Cek gap ke cue asli berikutnya
   780→   780→      // GAP di SRT baru = (cue[n+1].start di SRT baru) - (cue[n].end di SRT baru)
   781→   781→      //                = (nextEntry.start + offset) - newEnd
   782→   782→      //                = nextEntry.start - entry.start - cueAudio.durationSec  ← offset saling cancel
   783→   783→      // BUG SEBELUMNYA: gapToNext = nextEntry.start - newEnd
   784→   784→      //   → nextEntry.start (tanpa offset) vs newEnd (dengan offset) → cascade offset raksasa
   785→   785→      // FIX: hitung gapToNext tanpa offset effect (relative ke cue[n].originalStart)
   786→   786→      const nextEntry = entries[i + 1]
   787→   787→      if (nextEntry) {
   788→   788→        // gapToNext = jarak dari cue[n] newEnd ke cue[n+1] newStart di SRT BARU
   789→   789→        // = (nextEntry.originalStart + offset) - (entry.originalStart + offset + audioDur)
   790→   790→        // = nextEntry.originalStart - entry.originalStart - audioDur
   791→   791→        // (offset saling cancel, jadi tidak masalah berapapun offset saat ini)
   792→   792→        const cueToCueDistance = nextEntry.start - entry.start // jarak dari start cue[n] ke start cue[n+1] di SRT asli
   793→   793→        const gapToNext = cueToCueDistance - cueAudio.durationSec // sisa setelah audio cue[n]
   794→   794→
   795→   795→        if (gapToNext < minGapSec) {
   796→   796→          // Audio overflow → push back cue berikutnya
   797→   797→          // pushBack = berapa banyak cue[n+1] harus dimajukan ke depan
   798→   798→          // Kalau gapToNext = -3 (overflow 3s), pushBack = 0.15 - (-3) = 3.15
   799→   799→          const pushBack = minGapSec - gapToNext
   800→   800→          offset += pushBack
   801→   801→
   802→   802→          // Gap di retime map: dari newEnd (SRT baru) ke (newEnd + pushBack)
   803→   803→          // = pushBack detik hening
   804→   804→          retimePoints.push({
   805→   805→            cueIndex: i,
   806→   806→            originalStart: entry.end,
   807→   807→            originalEnd: nextEntry.start,
   808→   808→            newStart: newEnd,
   809→   809→            newEnd: newEnd + pushBack, // gap duration = pushBack (hening)
   810→   810→            originalDuration: nextEntry.start - entry.end,
   811→   811→            newDuration: pushBack,
   812→   812→            factor: pushBack / Math.max(0.001, nextEntry.start - entry.end),
   813→   813→            type: 'gap',
   814→   814→          })
   815→   815→        } else {
   816→   816→          // Gap cukup → tidak push back, offset tidak berubah
   817→   817→          // Gap di retime map = sisa gap asli (gapToNext)
   818→   818→          retimePoints.push({
   819→   819→            cueIndex: i,
   820→   820→            originalStart: entry.end,
   821→   821→            originalEnd: nextEntry.start,
   822→   822→            newStart: newEnd,
   823→   823→            newEnd: newEnd + gapToNext,
   824→   824→            originalDuration: nextEntry.start - entry.end,
   825→   825→            newDuration: gapToNext,
   826→   826→            factor: 1.0,
   827→   827→            type: 'gap',
   828→   828→          })
   829→   829→        }
   830→   830→      }
   831→   831→    }
   832→   832→    console.log('[Dubbing] Stitching step 1 done:', { indexedNewEntries: indexedNewEntries.length, offsetSec: offset })
   833→   833→
   834→   834→    // Step 2: Allocate audio buffer + mix
   835→   835→    const newDurationSec = indexedNewEntries.length > 0
   836→   836→      ? indexedNewEntries[indexedNewEntries.length - 1].end + 0.5
   837→   837→      : 0
   838→   838→    // Sanity check: kalau newDurationSec tidak finite atau > 12 jam, throw dengan message jelas
   839→   839→    // 12 jam = max wajar untuk SRT panjang yang di-dub natural (3 jam MP4 + 4x offset natural)
   840→   840→    if (!isFinite(newDurationSec) || newDurationSec <= 0) {
   841→   841→      throw new Error(`newDurationSec invalid: ${newDurationSec} (indexedNewEntries: ${indexedNewEntries.length}, last.end: ${indexedNewEntries.length > 0 ? indexedNewEntries[indexedNewEntries.length - 1].end : 'n/a'})`)
   842→   842→    }
   843→   843→    if (newDurationSec > 43200) {
   844→   844→      throw new Error(`newDurationSec ${newDurationSec}s > 12 jam — kemungkinan ada cue dengan duration raksasa (max 300s per cue)`)
   845→   845→    }
   846→   846→    const totalSamples = Math.floor(newDurationSec * OUTPUT_SAMPLE_RATE)
   847→   847→    console.log('[Dubbing] Stitching step 2: allocate buffer...', { newDurationSec, totalSamples, bytesMB: (totalSamples * 4 / 1024 / 1024).toFixed(1) })
   848→   848→    if (totalSamples <= 0 || !isFinite(totalSamples) || totalSamples > 2000000000) {
   849→   849→      throw new Error(`Invalid totalSamples: ${totalSamples} (newDurationSec=${newDurationSec})`)
   850→   850→    }
   851→   851→    const allAudio = new Float32Array(totalSamples)
   852→   852→
   853→   853→    for (const newEntry of indexedNewEntries) {
   854→   854→      const cueAudio = cueAudios[newEntry.cueIndex]
   855→   855→      const position = Math.floor(newEntry.start * OUTPUT_SAMPLE_RATE)
   856→   856→      if (cueAudio.pcm.length > 0 && position < totalSamples) {
   857→   857→        mixAudioInto(allAudio, cueAudio.pcm, position)
   858→   858→      }
   859→   859→    }
   860→   860→    console.log('[Dubbing] Stitching step 2 done: audio mixed')
   861→   861→
   862→   862→    // Step 3: Encode WAV
   863→   863→    console.log('[Dubbing] Stitching step 3: encode WAV...')
   864→   864→    blob = encodeWav(allAudio, OUTPUT_SAMPLE_RATE)
   865→   865→    audioDurationSec = allAudio.length / OUTPUT_SAMPLE_RATE
   866→   866→    console.log('[Dubbing] Stitching step 3 done:', { audioDurationSec, blobSize: blob.size })
   867→   867→
   868→   868→    // Step 4: Bangun SRT string
   869→   869→    console.log('[Dubbing] Stitching step 4: build SRT content...')
   870→   870→    srtContent = ''
   871→   871→    for (let i = 0; i < indexedNewEntries.length; i++) {
   872→   872→      const e = indexedNewEntries[i]
   873→   873→      srtContent += `${i + 1}\n`
   874→   874→      srtContent += `${formatTimeSrt(e.start)} --> ${formatTimeSrt(e.end)}\n`
   875→   875→      srtContent += e.text + '\n\n'
   876→   876→    }
   877→   877→    console.log('[Dubbing] Stitching step 4 done:', { srtLength: srtContent.length, cueCount: indexedNewEntries.length })
   878→   878→
   879→   879→    // Step 5: Bangun retime map
   880→   880→    console.log('[Dubbing] Stitching step 5: build retime map...')
   881→   881→    retimeMap = {
   882→   882→      version: '1.0',
   883→   883→      source: `Original SRT: ${entries.length} cues`,
   884→   884→      speed,
   885→   885→      minGapSec,
   886→   886→      totalOffsetSec: offset,
   887→   887→      originalDurationSec: entries[entries.length - 1].end,
   888→   888→      newDurationSec: audioDurationSec,
   889→   889→      points: retimePoints,
   890→   890→    }
   891→   891→    console.log('[Dubbing] Stitching step 5 done')
   892→   892→  } catch (stitchError) {
   893→   893→    const errMsg = `Gagal di tahap stitching (95%): ${(stitchError as Error).message}. Stack: ${(stitchError as Error).stack?.slice(0, 200)}`
   894→   894→    console.error('[Dubbing] ' + errMsg, stitchError)
   895→   895→    throw new Error(errMsg)
   896→   896→  }
   897→   897→
   898→   898→  opts.onStage?.({ stage: 'done', message: 'Dubbing selesai', percent: 100 })
   899→   899→
   900→   900→  return {
   901→   901→    audioBlob: blob,
   902→   902→    audioDurationSec,
   903→   903→    srtContent,
   904→   904→    retimeMap,
   905→   905→    retimeMapJson: JSON.stringify(retimeMap, null, 2),
   906→   906→    newEntries: indexedNewEntries.map(({ start, end, text }) => ({ start, end, text })),
   907→   907→  }
   908→   908→}

   910→   910→/**
   911→   911→ * Format waktu SRT: "HH:MM:SS,mmm"
   912→   912→ */
   913→   913→function formatTimeSrt(seconds: number): string {
   914→   914→  if (seconds < 0) seconds = 0
   915→   915→  const h = Math.floor(seconds / 3600)
   916→   916→  const m = Math.floor((seconds % 3600) / 60)
   917→   917→  const s = Math.floor(seconds % 60)
   918→   918→  let ms = Math.round((seconds - Math.floor(seconds)) * 1000)
   919→   919→  if (ms === 1000) {
   920→   920→    ms = 0
   921→   921→  }
   922→   922→  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')},${String(ms).padStart(3, '0')}`
   923→   923→}

   938→   938→export function formatDuration(sec: number): string {
   939→   939→  if (!isFinite(sec)) return '0s'
   940→   940→  const h = Math.floor(sec / 3600)
   941→   941→  const m = Math.floor((sec % 3600) / 60)
   942→   942→  const s = Math.floor(sec % 60)
   943→   943→  if (h > 0) return `${h}j ${m}m ${s}s`
   944→   944→  if (m > 0) return `${m}m ${s}s`
   945→   945→  return `${s}s`
   946→   946→}

   948→   948→export function formatBytes(bytes: number): string {
   949→   949→  if (bytes === 0) return '0 B'
   950→   950→  const k = 1024
   951→   951→  const units = ['B', 'KB', 'MB', 'GB', 'TB']
   952→   952→  const i = Math.floor(Math.log(bytes) / Math.log(k))
   953→   953→  return `${(bytes / Math.pow(k, i)).toFixed(1)} ${units[i]}`
   954→   954→}
