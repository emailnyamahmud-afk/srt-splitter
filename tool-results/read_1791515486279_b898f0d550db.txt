    61→def load_env_file():
    62→    """Load .env file dari ~/Dubbing/.env (kalau ada).
    63→    Supaya user tidak perlu set env vars manual tiap kali update kamus-tui.py.
    64→    Format .env:
    65→      NEXT_PUBLIC_SUPABASE_URL=https://xxx.supabase.co
    66→      NEXT_PUBLIC_SUPABASE_ANON_KEY=eyJxxx...
    67→    """
    68→    if not ENV_FILE.exists():
    69→        return False
    70→    try:
    71→        with open(ENV_FILE, 'r', encoding='utf-8') as f:
    72→            for line in f:
    73→                line = line.strip()
    74→                # Skip comment dan empty line
    75→                if not line or line.startswith('#'):
    76→                    continue
    77→                # Parse KEY=VALUE
    78→                if '=' in line:
    79→                    key, value = line.split('=', 1)
    80→                    key = key.strip()
    81→                    value = value.strip().strip('"').strip("'")
    82→                    # Hanya set kalau belum ada di os.environ (os.environ lebih prioritas)
    83→                    if key and key not in os.environ:
    84→                        os.environ[key] = value
    85→        return True
    86→    except Exception as e:
    87→        print(f'  ⚠ Gagal load .env: {e}')
    88→        return False

   126→def save_kamus(data):
   127→    """Save ke JSON file + recompute status untuk SEMUA entries.
   128→
   129→    Status logic:
   130→    - 'ready' = ngoko + krama + arti SEMUA terisi → siap upload ke Supabase
   131→    - 'draft' = belum lengkap
   132→
   133→    SELALU recompute status setiap save (jangan skip yang sudah punya status).
   134→    User edit arti → status harus recompute dari draft → ready.
   135→
   136→    entry_id: dipertahankan dari build-kamus-bersih (urut alfabetis).
   137→    Kalau ada entry yang di-delete (via merge), re-number entry_id 1..N supaya tetap konsisten.
   138→    """
   139→    for w in data['words']:
   140→        ngoko = (w.get('ngoko') or '').strip()
   141→        krama = (w.get('krama') or '').strip()
   142→        arti = (w.get('arti') or '').strip()
   143→        w['status'] = 'ready' if (ngoko and krama and arti) else 'draft'
   144→    # Re-number entry_id (urut posisi list = urut alfabetis dari build-kamus-bersih)
   145→    for i, w in enumerate(data['words'], 1):
   146→        w['entry_id'] = i
   147→    path = get_kamus_path()
   148→    with open(path, 'w', encoding='utf-8') as f:
   149→        json.dump(data, f, ensure_ascii=False, indent=2)

   310→def browse_list(data, entries_with_idx, title):
   311→    """Browse list of entries (idx, entry) dengan pagination"""
   312→    total = len(entries_with_idx)
   313→    if total == 0:

   315→        input('\n  Tekan Enter...')
   316→        return
   317→
   318→    page_size = 20
   319→    current_page = 0
   320→
   321→    while True:
   322→        start = current_page * page_size
   323→        end = min(start + page_size, total)
   324→        page = entries_with_idx[start:end]
   325→
   326→        os.system('clear' if os.name != 'nt' else 'cls')
   327→        print('╔' + '═' * 60 + '╗')
   328→        print(f'║  {title[:58]}' + ' ' * max(0, 58 - len(title)) + '║')
   329→        print(f'║  {total} entri | Hal {current_page + 1}/{(total + page_size - 1) // page_size}' + ' ' * max(0, 30 - len(str(total)) - len(str(current_page + 1)) - len(str((total + page_size - 1) // page_size))) + '║')
   330→        print('╚' + '═' * 60 + '╝')
   331→        print()
   332→
   333→        # Build label → idx mapping (untuk lookup saat select)
   334→        label_to_idx = {}
   335→        choices = []
   336→        for orig_idx, entry in page:
   337→            ngoko = entry.get('ngoko', '') or entry.get('krama', '') or ''
   338→            ngoko_display = ngoko[:25]
   339→            krama = entry.get('krama', '') or ''
   340→            ki = entry.get('krama_inggil', '') or ''
   341→            arti = entry.get('arti', '') or ''
   342→            status = entry.get('status', 'draft')
   343→            register = entry.get('register', 'umum')
   344→            entry_id = entry.get('entry_id', orig_idx + 1)
   345→            icon = '✓' if status == 'ready' else '○'

   350→                label += f' → {krama[:15]:15s}'
   351→            if ki:
   352→                label += f' | ki: {ki[:10]}'
   353→            if arti:
   354→                label += f' | {arti[:15]}'

   356→            if register != 'umum':
   357→                label += f' [{register}]'
   358→            label_to_idx[label] = orig_idx
   359→            choices.append(label)
   360→
   361→        # Navigation
   362→        nav = []
   363→        if current_page > 0:
   364→            nav.append('◀ Halaman sebelumnya')

   366→            nav.append('▶ Halaman berikutnya')
   367→        nav.append('🔍 Search di list ini')

   369→

   371→
   372→        selected = questionary.select(
   373→            f'Pilih entri (✓=sudah diedit, ○=belum):',
   374→            choices=choices,
   375→            default=choices[0],
   376→        ).ask()
   377→
   378→        if not selected or 'Kembali ke menu' in selected:
   379→            return
   380→
   381→        if 'sebelumnya' in selected:
   382→            current_page = max(0, current_page - 1)
   383→            continue
   384→

   386→            current_page += 1

   389→        if 'Search' in selected:
   390→            search_query = questionary.text('Cari kata di list ini:').ask()

   392→                search_lower = search_query.lower().strip()
   393→                # Exact match dulu, baru substring (sama seperti search_menu)

   398→                    krama = (e.get('krama', '') or '').lower()

   401→                    if ngoko == search_lower or krama == search_lower or ki == search_lower or arti == search_lower:
   402→                        exact_f.append((i, e))
   403→                    elif (search_lower in ngoko or search_lower in krama or
   404→                          search_lower in ki or search_lower in arti):
   405→                        substr_f.append((i, e))
   406→                filtered = exact_f + substr_f
   407→                if filtered:
   408→                    browse_list(data, filtered, f'🔍 Search "{search_query}"')
   409→                else:
   410→                    print(f'\n  Tidak ada hasil untuk "{search_query}"')
   411→                    input('\n  Tekan Enter...')
   412→            continue
   413→
   414→        # Find which entry was selected (lookup di label_to_idx)
   415→        match_idx = label_to_idx.get(selected)
   416→        if match_idx is not None:
   417→            edit_entry(data, match_idx)

   420→def search_menu(data):
   421→    """Search kamus — exact match dulu, baru substring. Hanya di ngoko/krama/krama_inggil/arti."""
   422→    os.system('clear' if os.name != 'nt' else 'cls')
   423→    print('╔' + '═' * 60 + '╗')
   424→    print('║  🔍 Search Kamus' + ' ' * 44 + '║')
   425→    print('╚' + '═' * 60 + '╝')
   426→    print()
   427→
   428→    query = questionary.text('Cari kata (di ngoko / krama / arti):').ask()
   429→    if not query:
   430→        return
   431→
   432→    search_lower = query.lower().strip()
   433→    # Exact match dulu (ngoko/krama/krama_inggil/arti persis = kata), baru substring
   434→    # TIDAK search di keterangan (terlalu banyak noise)
   435→    exact = []
   436→    substr = []
   437→    for i, entry in enumerate(data['words']):
   438→        ngoko = (entry.get('ngoko') or '').lower()
   439→        krama = (entry.get('krama') or '').lower()
   440→        ki = (entry.get('krama_inggil') or '').lower()
   441→        arti = (entry.get('arti') or '').lower()
   442→        if ngoko == search_lower or krama == search_lower or ki == search_lower or arti == search_lower:
   443→            exact.append((i, entry))
   444→        elif (search_lower in ngoko or search_lower in krama or
   445→              search_lower in ki or search_lower in arti):
   446→            substr.append((i, entry))
   447→    matches = exact + substr
   448→
   449→    if not matches:
   450→        print(f'\n  Tidak ada hasil untuk "{query}"')
   451→        input('\n  Tekan Enter...')
   452→        return
   453→
   454→    browse_list(data, matches, f'🔍 Search "{query}"')
