     4→import {

     8→  Download,
     9→  Clock,

    18→} from 'lucide-react'
    19→import { Button } from '@/components/ui/button'
    20→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card'
    21→import { Input } from '@/components/ui/input'
    22→import { Label } from '@/components/ui/label'
    23→import { Switch } from '@/components/ui/switch'
    24→import { Slider } from '@/components/ui/slider'
    25→import { Badge } from '@/components/ui/badge'
    26→import { Separator } from '@/components/ui/separator'
    27→import { ScrollArea } from '@/components/ui/scroll-area'
    28→import { toast } from 'sonner'
    29→import { TtsPanel } from '@/components/tts-panel'
    30→import { TtsTextPanel } from '@/components/tts-text-panel'
    31→import { TranslatePanel } from '@/components/translate-panel'
    32→import { RapikanJawaPanel } from '@/components/rapikan-jawa-panel'
    33→import { KamusEditorPanel } from '@/components/kamus-editor-panel'
    34→import { DualSrtEditor } from '@/components/srt-editor-panel'
    35→import {
    36→  parseSrt,
    37→  splitEntries,
    38→  serializePart,
    39→  downloadTextFile,
    40→  downloadAllPartsAsZip,
    41→  type SrtEntry,
    42→  type SplitResult,
    43→} from '@/lib/srt'
    44→
    45→export default function Home() {
    46→  // Persist state to localStorage so refresh doesn't reset upload
    47→  const STORAGE_KEY = 'srt-splitter-state-v1'
    48→
    49→  const [file, setFile] = useState<File | null>(null)
    50→  const [fileContent, setFileContent] = useState<string>('') // SRT text persisted
    51→  const [entries, setEntries] = useState<SrtEntry[]>([])
    52→  const [fileName, setFileName] = useState<string>('')
    53→  const [maxMinutes, setMaxMinutes] = useState<number>(30)
    54→  const [splitMode, setSplitMode] = useState<'duration' | 'chars'>('duration')
    55→  const [maxChars, setMaxChars] = useState<number>(4500)
    56→  const [prefix, setPrefix] = useState<string>('S6')
    57→  const [resetTimestamps, setResetTimestamps] = useState<boolean>(false)
    58→  const [isDragging, setIsDragging] = useState<boolean>(false)
    59→  const [isZipping, setIsZipping] = useState<boolean>(false)
    60→  const [isListExpanded, setIsListExpanded] = useState<boolean>(false)
    61→  const [isPreviewExpanded, setIsPreviewExpanded] = useState<boolean>(false)
    62→  const fileInputRef = useRef<HTMLInputElement>(null)
    63→
    64→  // Restore state from localStorage on mount
    65→  useEffect(() => {
    66→    try {
    67→      const saved = localStorage.getItem(STORAGE_KEY)
    68→      if (!saved) return
    69→      const data = JSON.parse(saved) as {
    70→        fileName?: string
    71→        fileContent?: string
    72→        maxMinutes?: number
    73→        prefix?: string
    74→        resetTimestamps?: boolean
    75→      }
    76→      if (data.fileContent && data.fileName) {
    77→        const parsed = parseSrt(data.fileContent)
    78→        if (parsed.length > 0) {
    79→          setEntries(parsed)
    80→          setFileName(data.fileName)
    81→          setFileContent(data.fileContent)
    82→          // Reconstruct a pseudo-File object so UI shows file as loaded
    83→          try {
    84→            const restoredFile = new File([data.fileContent], data.fileName, { type: 'application/x-subrip' })
    85→            setFile(restoredFile)
    86→          } catch {
    87→            // File constructor might fail in some environments; entries+fileName is enough
    88→          }
    89→          if (data.maxMinutes) setMaxMinutes(data.maxMinutes)
    90→          if (data.prefix) setPrefix(data.prefix)
    91→          if (typeof data.resetTimestamps === 'boolean') setResetTimestamps(data.resetTimestamps)
    92→          toast.info(`Restored ${parsed.length} subtitle dari sesi sebelumnya.`)
    93→        }
    94→      }
    95→    } catch (e) {
    96→      console.warn('Failed to restore state from localStorage:', e)
    97→    }
    98→  }, [])
    99→
   100→  // Save state to localStorage whenever relevant state changes
   101→  useEffect(() => {
   102→    if (!fileContent || !fileName) {
   103→      // Clear storage if no file
   104→      localStorage.removeItem(STORAGE_KEY)
   105→      return
   106→    }
   107→    try {
   108→      const data = { fileName, fileContent, maxMinutes, prefix, resetTimestamps }
   109→      localStorage.setItem(STORAGE_KEY, JSON.stringify(data))
   110→    } catch (e) {
   111→      console.warn('Failed to save state to localStorage:', e)
   112→      // localStorage might be full (large SRT files)
   113→    }
   114→  }, [fileName, fileContent, maxMinutes, prefix, resetTimestamps])
   115→
   116→  const splitResult: SplitResult | null = useMemo(() => {
   117→    if (entries.length === 0) return null
   118→    return splitEntries(entries, {
   119→      maxMinutes,
   120→      prefix,
   121→      resetTimestamps,
   122→      splitOnBoundary: true,
   123→      splitByChars: splitMode === 'chars' ? maxChars : undefined,
   124→    })
   125→  }, [entries, maxMinutes, prefix, resetTimestamps, splitMode, maxChars])
   126→
   127→  const handleFile = useCallback(async (f: File) => {
   128→    if (!f.name.toLowerCase().endsWith('.srt')) {
   129→      toast.error('File harus berekstensi .srt')
   130→      return
   131→    }
   132→    try {
   133→      const text = await f.text()
   134→      const parsed = parseSrt(text)
   135→      if (parsed.length === 0) {
   136→        toast.error('Tidak ada subtitle valid ditemukan di file.')
   137→        return
   138→      }
   139→      setFile(f)
   140→      setEntries(parsed)
   141→      setFileContent(text) // persist to localStorage
   142→      // Auto-fill prefix from filename if it looks like a season code
   143→      const baseName = f.name.replace(/\.srt$/i, '').replace(/\[.*?\]/g, '').trim()
   144→      setPrefix(baseName.slice(0, 6) || 'S6')
   145→      setFileName(f.name)
   146→      toast.success(`Berhasil parsing ${parsed.length} subtitle.`)
   147→    } catch (e) {
   148→      console.error(e)
   149→      toast.error('Gagal membaca file. Pastikan encoding UTF-8.')
   150→    }
   151→  }, [])
   152→
   153→  const onDrop = useCallback(
   154→    (e: React.DragEvent<HTMLDivElement>) => {
   155→      e.preventDefault()
   156→      setIsDragging(false)
   157→      const f = e.dataTransfer.files?.[0]
   158→      if (f) handleFile(f)
   159→    },
   160→    [handleFile],
   161→  )
   162→
   163→  const onInputChange = useCallback(
   164→    (e: React.ChangeEvent<HTMLInputElement>) => {
   165→      const f = e.target.files?.[0]
   166→      if (f) handleFile(f)
   167→    },
   168→    [handleFile],
   169→  )
   170→
   171→  const formatHMS = (sec: number): string => {
   172→    const h = Math.floor(sec / 3600)
   173→    const m = Math.floor((sec % 3600) / 60)
   174→    const s = Math.floor(sec % 60)
   175→    if (h > 0) return `${h}j ${m}m ${s}s`
   176→    return `${m}m ${s}s`
   177→  }
   178→
   179→  const formatTS = (sec: number): string => {
   180→    const h = Math.floor(sec / 3600)
   181→    const m = Math.floor((sec % 3600) / 60)
   182→    const s = Math.floor(sec % 60)
   183→    return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   184→  }
   185→
   186→  const reset = () => {
   187→    setFile(null)
   188→    setEntries([])
   189→    setFileName('')
   190→    setFileContent('') // triggers localStorage clear via useEffect
   191→    if (fileInputRef.current) fileInputRef.current.value = ''
   192→  }
   193→
   194→  const totalDuration = entries.length > 0 ? entries[entries.length - 1].end : 0
   195→
   196→  return (
   197→    <div className="min-h-screen flex flex-col bg-gradient-to-br from-slate-50 to-slate-100 dark:from-slate-950 dark:to-slate-900">
   198→      <header className="border-b bg-white/80 dark:bg-slate-900/80 backdrop-blur-sm sticky top-0 z-10">
   199→        <div className="container mx-auto max-w-5xl px-4 py-3 flex items-center justify-between">
   200→          <div className="flex items-center gap-2.5">
   201→            <img src="/logo.svg" alt="SRT Splitter" className="size-8 rounded-lg" />
   202→            <div>
   203→              <h1 className="text-lg font-bold tracking-tight">SRT Splitter</h1>
   204→              <p className="text-[11px] text-muted-foreground">Dubbing Studio — Split, Translate, TTS, Editor Jawa</p>
   205→            </div>
   206→          </div>
   207→          <Badge variant="secondary" className="hidden sm:flex">
   208→            <CheckCircle2 className="size-3 mr-1" /> 100% Sync
   209→          </Badge>
   210→        </div>
   211→      </header>
   212→
   213→      <main className="container mx-auto max-w-5xl px-4 py-8 flex-1 space-y-6">
   214→        {/* Upload area */}
   215→        {!file && (
   216→          <Card className="border-2 border-dashed hover:border-primary/50 transition-colors">
   217→            <CardContent
   218→              className="py-16 px-6 flex flex-col items-center justify-center text-center cursor-pointer"
   219→              onClick={() => fileInputRef.current?.click()}
   220→              onDragOver={(e) => {
   221→                e.preventDefault()
   222→                setIsDragging(true)
   223→              }}
   224→              onDragLeave={() => setIsDragging(false)}
   225→              onDrop={onDrop}
   226→            >
   227→              <div
   228→                className={`size-20 rounded-full bg-gradient-to-br from-amber-100 to-orange-100 dark:from-amber-950/40 dark:to-orange-950/40 flex items-center justify-center mb-6 transition-transform ${
   229→                  isDragging ? 'scale-110' : ''
   230→                }`}
   231→              >
   232→                <Upload className="size-8 text-amber-600" />
   233→              </div>
   234→              <h2 className="text-2xl font-semibold mb-2">Drop file .srt di sini</h2>
   235→              <p className="text-muted-foreground mb-4 max-w-md">
   236→                Klik untuk pilih file atau seret-dan-jatuhkan. Semua proses dilakukan di
   237→                browser Anda — file tidak pernah dikirim ke server mana pun.
   238→              </p>
   239→              <Button size="lg">
   240→                <FileText className="size-4 mr-2" /> Pilih File SRT
   241→              </Button>
   242→              <input
   243→                ref={fileInputRef}
   244→                type="file"
   245→                accept=".srt"
   246→                className="hidden"
   247→                onChange={onInputChange}
   248→              />
   249→              <div className="mt-8 grid grid-cols-1 sm:grid-cols-3 gap-3 max-w-2xl text-left">
   250→                <div className="flex items-start gap-2 text-sm">
   251→                  <Clock className="size-4 text-amber-600 mt-0.5 shrink-0" />
   252→                  <span>Bisa potong per 30 menit atau berapapun sesuai kebutuhan</span>
   253→                </div>
   254→                <div className="flex items-start gap-2 text-sm">
   255→                  <Hash className="size-4 text-amber-600 mt-0.5 shrink-0" />
   256→                  <span>Timestamp bisa dipertahankan atau direset ke 00:00:00</span>
   257→                </div>
   258→                <div className="flex items-start gap-2 text-sm">
   259→                  <Film className="size-4 text-amber-600 mt-0.5 shrink-0" />
   260→                  <span>Dipotong di batas subtitle, tidak ada teks yang rusak</span>
   261→                </div>
   262→              </div>
   263→            </CardContent>
   264→          </Card>
   265→        )}
   266→
   267→        {/* Main workspace */}
   268→        {file && splitResult && (
   269→          <>
   270→            {/* File info + reset */}
   271→            <Card>
   272→              <CardHeader className="pb-4">
   273→                <div className="flex items-start justify-between gap-3">
   274→                  <div className="min-w-0 flex-1">
   275→                    <CardTitle className="text-base truncate">{fileName}</CardTitle>
   276→                    <CardDescription className="mt-1 flex flex-wrap items-center gap-3">
   277→                      <span className="flex items-center gap-1">
   278→                        <Hash className="size-3" /> {entries.length} subtitle
   279→                      </span>
   280→                      <span className="flex items-center gap-1">
   281→                        <Clock className="size-3" /> {formatHMS(totalDuration)}
   282→                      </span>
   283→                    </CardDescription>
   284→                  </div>
   285→                  <Button variant="ghost" size="sm" onClick={reset}>
   286→                    <RefreshCw className="size-4 mr-1" /> File lain
   287→                  </Button>
   288→                </div>
   289→              </CardHeader>
   290→            </Card>
   291→
   292→            {/* Settings */}
   293→            <Card>
   294→              <CardHeader>
   295→                <CardTitle className="text-lg">Pengaturan Split</CardTitle>
   296→                <CardDescription>
   297→                  Atur durasi tiap file output dan opsi timestamp.
   298→                </CardDescription>
   299→              </CardHeader>
   300→              <CardContent className="space-y-6">
   301→                {/* Split mode selector */}
   302→                <div className="space-y-2">
   303→                  <Label className="text-sm font-medium">Mode split</Label>
   304→                  <div className="flex gap-2">
   305→                    <Button
   306→                      size="sm"
   307→                      variant={splitMode === 'duration' ? 'default' : 'outline'}
   308→                      onClick={() => setSplitMode('duration')}
   309→                      className="h-8"
   310→                    >
   311→                      By Durasi
   312→                    </Button>
   313→                    <Button
   314→                      size="sm"
   315→                      variant={splitMode === 'chars' ? 'default' : 'outline'}
   316→                      onClick={() => setSplitMode('chars')}
   317→                      className="h-8"
   318→                    >
   319→                      By Karakter
   320→                    </Button>
   321→                  </div>
   322→                </div>
   323→
   324→                {/* Duration slider (only if splitMode = 'duration') */}
   325→                {splitMode === 'duration' && (
   326→                <div className="space-y-3">
   327→                  <div className="flex items-center justify-between">
   328→                    <Label className="text-sm font-medium">Durasi maksimal per file</Label>
   329→                    <Badge variant="outline" className="font-mono">
   330→                      {maxMinutes >= 60
   331→                        ? `${Math.floor(maxMinutes / 60)}j ${maxMinutes % 60 > 0 ? `${maxMinutes % 60}m` : ''}`
   332→                        : `${maxMinutes} menit`}
   333→                    </Badge>
   334→                  </div>
   335→                  <Slider
   336→                    value={[maxMinutes]}
   337→                    onValueChange={(v) => setMaxMinutes(v[0])}
   338→                    min={5}
   339→                    max={300}
   340→                    step={5}
   341→                    className="w-full"
   342→                  />
   343→                  <div className="text-xs text-muted-foreground text-center">
   344→                    5 menit → 5 jam. Untuk film 3-4 jam, pilih 240m (4j) atau 300m (5j) agar tidak di-split.
   345→                  </div>
   346→                  <div className="flex flex-wrap gap-2">
   347→                    {[10, 15, 20, 30, 45, 60, 90, 120, 180, 240, 300].map((m) => {
   348→                      const label = m >= 60
   349→                        ? (m % 60 === 0 ? `${m / 60}j` : `${Math.floor(m / 60)}j${m % 60}m`)
   350→                        : `${m}m`
   351→                      return (
   352→                        <Button
   353→                          key={m}
   354→                          size="sm"
   355→                          variant={maxMinutes === m ? 'default' : 'outline'}
   356→                          onClick={() => setMaxMinutes(m)}
   357→                          className="h-7 px-2 text-xs"
   358→                        >
   359→                          {label}
   360→                        </Button>
   361→                      )
   362→                    })}
   363→                  </div>
   364→                </div>
   365→                )}
   366→
   367→                {/* Character count (only if splitMode = 'chars') */}
   368→                {splitMode === 'chars' && (
   369→                <div className="space-y-3">
   370→                  <div className="flex items-center justify-between">
   371→                    <Label className="text-sm font-medium">Maksimal karakter per file</Label>
   372→                    <Badge variant="outline" className="font-mono">
   373→                      {maxChars} karakter
   374→                    </Badge>
   375→                  </div>
   376→                  <Slider
   377→                    value={[maxChars]}
   378→                    onValueChange={(v) => setMaxChars(v[0])}
   379→                    min={1000}
   380→                    max={5000}
   381→                    step={500}
   382→                    className="w-full"
   383→                  />
   384→                  <div className="text-xs text-muted-foreground text-center">
   385→                    Cocok untuk Edge TTS (limit 5000 karakter per request). 4500 karakter ≈ 3-5 menit audio.
   386→                  </div>
   387→                  <div className="flex flex-wrap gap-2">
   388→                    {[2000, 3000, 4500, 5000].map((c) => (
   389→                      <Button
   390→                        key={c}
   391→                        size="sm"
   392→                        variant={maxChars === c ? 'default' : 'outline'}
   393→                        onClick={() => setMaxChars(c)}
   394→                        className="h-7 px-2 text-xs"
   395→                      >
   396→                        {c} chars
   397→                      </Button>
   398→                    ))}
   399→                  </div>
   400→                </div>
   401→                )}
   402→
   403→                <Separator />
   404→
   405→                {/* Prefix */}
   406→                <div className="space-y-2">
   407→                  <Label htmlFor="prefix" className="text-sm font-medium">
   408→                    Prefix nama file
   409→                  </Label>
   410→                  <div className="flex items-center gap-2">
   411→                    <Input
   412→                      id="prefix"
   413→                      value={prefix}
   414→                      onChange={(e) => setPrefix(e.target.value.replace(/[^\w-]/g, ''))}
   415→                      placeholder="S6"
   416→                      className="max-w-[200px]"
   417→                    />
   418→                    <span className="text-sm text-muted-foreground font-mono">
   419→                      → {prefix}-01.srt, {prefix}-02.srt, …
   420→                    </span>
   421→                  </div>
   422→                </div>
   423→
   424→                <Separator />
   425→
   426→                {/* Reset timestamps */}
   427→                <div className="flex items-start justify-between gap-3">
   428→                  <div>
   429→                    <Label htmlFor="reset-ts" className="text-sm font-medium">
   430→                      Reset timestamp per file ke 00:00:00
   431→                    </Label>
   432→                    <p className="text-xs text-muted-foreground mt-1">
   433→                      Matikan untuk mempertahankan timestamp asli (cocok untuk dipasang
   434→                      langsung ke video asli tanpa offset).
   435→                    </p>
   436→                  </div>
   437→                  <Switch
   438→                    id="reset-ts"
   439→                    checked={resetTimestamps}
   440→                    onCheckedChange={setResetTimestamps}
   441→                  />
   442→                </div>
   443→              </CardContent>
   444→            </Card>
   445→
   446→            {/* Result overview — collapsible (default collapsed to save space) */}
   447→            <Card>
   448→              <CardHeader>
   449→                <div className="flex items-center justify-between gap-3">
   450→                  <button
   451→                    type="button"
   452→                    onClick={() => setIsListExpanded(!isListExpanded)}
   453→                    className="flex items-center gap-2 text-left hover:opacity-80 transition-opacity"
   454→                    aria-expanded={isListExpanded}
   455→                  >
   456→                    {isListExpanded ? (
   457→                      <ChevronUp className="size-4 text-muted-foreground" />
   458→                    ) : (
   459→                      <ChevronDown className="size-4 text-muted-foreground" />
   460→                    )}
   461→                    <div>
   462→                      <CardTitle className="text-lg">Hasil Pemecahan</CardTitle>
   463→                      <CardDescription>
   464→                        {splitResult.parts.length} file — dipotong di batas subtitle.{' '}
   465→                        <span className="text-xs text-muted-foreground">
   466→                          ({isListExpanded ? 'Klik untuk sembunyikan' : 'Klik untuk melihat list'})
   467→                        </span>
   468→                      </CardDescription>
   469→                    </div>
   470→                  </button>
   471→                  <Button
   472→                    size="sm"
   473→                    disabled={isZipping}
   474→                    onClick={async () => {
   475→                      if (!splitResult) return
   476→                      setIsZipping(true)
   477→                      const tid = toast.loading('Membuat ZIP…')
   478→                      try {
   479→                        await downloadAllPartsAsZip(
   480→                          splitResult.parts,
   481→                          prefix,
   482→                          resetTimestamps,
   483→                        )
   484→                        toast.success(`Mengunduh ${prefix}-split.zip`, {
   485→                          id: tid,
   486→                        })
   487→                      } catch (e) {
   488→                        console.error(e)
   489→                        toast.error('Gagal membuat ZIP', { id: tid })
   490→                      } finally {
   491→                        setIsZipping(false)
   492→                      }
   493→                    }}
   494→                  >
   495→                    {isZipping ? (
   496→                      <>
   497→                        <RefreshCw className="size-4 mr-1 animate-spin" /> Mengemas…
   498→                      </>
   499→                    ) : (
   500→                      <>
   501→                        <Download className="size-4 mr-1" /> Unduh ZIP
   502→                      </>
   503→                    )}
   504→                  </Button>
   505→                </div>
   506→              </CardHeader>
   507→              {isListExpanded && (
   508→                <CardContent>
   509→                  <ScrollArea className="max-h-[480px] pr-3">
   510→                  <div className="space-y-2">
   511→                    {splitResult.parts.map((part) => {
   512→                      const filename = `${prefix}-${String(part.index).padStart(2, '0')}.srt`
   513→                      return (
   514→                        <div
   515→                          key={part.index}
   516→                          className="flex items-center gap-3 p-3 rounded-lg border bg-card hover:shadow-sm transition-shadow"
   517→                        >
   518→                          <div className="size-10 rounded-md bg-gradient-to-br from-amber-100 to-orange-100 dark:from-amber-950/40 dark:to-orange-950/40 flex items-center justify-center shrink-0">
   519→                            <span className="text-sm font-bold text-amber-700 dark:text-amber-400">
   520→                              {String(part.index).padStart(2, '0')}
   521→                            </span>
   522→                          </div>
   523→                          <div className="min-w-0 flex-1">
   524→                            <div className="flex items-center gap-2 flex-wrap">
   525→                              <span className="font-mono text-sm font-medium truncate">
   526→                                {filename}
   527→                              </span>
   528→                              <Badge variant="secondary" className="text-xs">
   529→                                {part.entryCount} subtitle
   530→                              </Badge>
   531→                            </div>
   532→                            <div className="text-xs text-muted-foreground mt-0.5 font-mono">
   533→                              {formatTS(part.startSec)} → {formatTS(part.endSec)}
   534→                              <span className="mx-1.5">•</span>
   535→                              durasi {formatHMS(part.durationSec)}

   696→
   697→function PreviewPart({
   698→  part,
   699→  resetTimestamps,
   700→}: {
   701→  part: SplitResult['parts'][number]
   702→  resetTimestamps: boolean
   703→}) {
   704→  const offset = resetTimestamps ? part.startSec : 0
   705→  const firstThree = part.entries.slice(0, 3)
   706→  const lastThree = part.entries.slice(-3)
   707→
   708→  const fmt = (sec: number) => {
   709→    const s = Math.max(0, sec - offset)
   710→    const h = Math.floor(s / 3600)
   711→    const m = Math.floor((s % 3600) / 60)
   712→    const sec2 = Math.floor(s % 60)
   713→    const ms = Math.round((s - Math.floor(s)) * 1000)
   714→    return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(sec2).padStart(2, '0')},${String(ms).padStart(3, '0')}`
   715→  }
   716→
   717→  const renderItem = (entry: SrtEntry, idx: number) => (
   718→    <div key={idx} className="text-sm">
   719→      <div className="font-mono text-xs text-muted-foreground mb-0.5">
   720→        {fmt(entry.start)} → {fmt(entry.end)}
   721→      </div>
   722→      <div>{entry.textLines.join(' ')}</div>
   723→    </div>
   724→  )
   725→
   726→  return (
   727→    <div className="grid gap-4 md:grid-cols-2">
   728→      <div>
   729→        <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">
   730→          Awal file
   731→        </h4>
   732→        <div className="space-y-3">{firstThree.map(renderItem)}</div>
   733→      </div>
   734→      <div>
   735→        <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">
   736→          Akhir file
   737→        </h4>
   738→        <div className="space-y-3">{lastThree.map(renderItem)}</div>
   739→      </div>
   740→    </div>
   741→  )
   742→}
