    81→def parse_srt(content: str):
    82→    """Parse SRT string jadi list of {index, start, end, text}."""
    83→    normalized = content.replace('\r\n', '\n').replace('\r', '\n').strip()
    84→    if not normalized:
    85→        return []
    86→    blocks = re.split(r'\n\s*\n', normalized)
    87→    entries = []
    88→    for block in blocks:
    89→        lines = [l for l in block.split('\n') if l.strip()]
    90→        if not lines:
    91→            continue
    92→        time_idx = -1
    93→        for i, line in enumerate(lines):
    94→            if '-->' in line:
    95→                time_idx = i
    96→                break
    97→        if time_idx == -1:
    98→            continue
    99→        m = re.match(
   100→            r'(\d{1,2}:\d{2}:\d{2}[,.]\d{1,3})\s*-->\s*(\d{1,2}:\d{2}:\d{2}[,.]\d{1,3})',
   101→            lines[time_idx],
   102→        )
   103→        if not m:
   104→            continue
   105→        start = parse_time(m.group(1))
   106→        end = parse_time(m.group(2))
   107→        text = '\n'.join(lines[time_idx + 1:])
   108→        entries.append({
   109→            'index': len(entries) + 1,
   110→            'start': start,
   111→            'end': end,
   112→            'text': text,
   113→        })
   114→    return entries

   116→
   117→def parse_time(ts: str) -> float:
   118→    """Parse 'HH:MM:SS,mmm' → seconds (float)."""
   119→    ts = ts.replace('.', ',')
   120→    hms, ms = ts.split(',') if ',' in ts else (ts, '0')
   121→    h, m, s = hms.split(':')
   122→    return int(h) * 3600 + int(m) * 60 + int(s) + int(ms) / 1000
   123→
   124→
   125→def format_time_ffmpeg(seconds: float) -> str:
   126→    """Format seconds → 'HH:MM:SS.mmm' (untuk ffmpeg trim)."""
   127→    if seconds < 0:
   128→        seconds = 0
   129→    h = int(seconds // 3600)
   130→    m = int((seconds % 3600) // 60)
   131→    s = int(seconds % 60)
   132→    ms = int((seconds - int(seconds)) * 1000)
   133→    return f'{h:02d}:{m:02d}:{s:02d}.{ms:03d}'

   140→def find_ffmpeg(custom_path: str = None) -> str:
   141→    if custom_path:
   142→        if os.path.isfile(custom_path) and os.access(custom_path, os.X_OK):
   143→            return custom_path
   144→        raise FileNotFoundError(f'ffmpeg tidak ditemukan di: {custom_path}')
   145→    path = shutil.which('ffmpeg') or shutil.which('ffmpeg.exe')
   146→    if path:
   147→        return path
   148→    for loc in ['/usr/local/bin/ffmpeg', '/usr/bin/ffmpeg', '/opt/homebrew/bin/ffmpeg',
   149→                '/Applications/ffmpeg', 'C:\\ffmpeg\\bin\\ffmpeg.exe']:
   150→        if os.path.isfile(loc):
   151→            return loc
   152→    raise FileNotFoundError('ffmpeg tidak ditemukan di PATH. Install ffmpeg atau specify via --ffmpeg.')

   169→def get_mp4_duration(ffprobe: str, mp4_path: str) -> float:
   170→    result = subprocess.run([
   171→        ffprobe, '-v', 'error',
   172→        '-show_entries', 'format=duration',
   173→        '-of', 'default=noprint_wrappers=1:nokey=1',
   174→        mp4_path,
   175→    ], capture_output=True, text=True)
   176→    return float(result.stdout.strip())

   179→def get_mp4_audio_streams(ffprobe: str, mp4_path: str) -> list:
   180→    result = subprocess.run([
   181→        ffprobe, '-v', 'error',
   182→        '-show_entries', 'stream=index,codec_type,codec_name,channels',
   183→        '-of', 'json',
   184→        mp4_path,
   185→    ], capture_output=True, text=True)
   186→    data = json.loads(result.stdout)
   187→    return [s for s in data.get('streams', []) if s.get('codec_type') == 'audio']

   190→# ============================================================
   191→# Build segment tasks

   193→
   194→def build_segment_tasks(mandarin_entries, jawa_entries, mp4_duration):
   195→    """
   196→    Build list of segment tasks untuk Pass 1.
   197→    Setiap task = 1 FFmpeg command untuk render 1 segment (cue atau gap).
   198→
   199→    Returns:
   200→      list of dict: {
   201→        'type': 'cue' | 'gap' | 'tail',
   202→        'index': int,
   203→        'mp4_start': float,  # start di MP4 ori
   204→        'mp4_end': float,    # end di MP4 ori
   205→        'target_duration': float,  # durasi setelah retim
   206→        'factor': float,     # target_duration / (mp4_end - mp4_start)
   207→        'output_file': str,  # path ke segment file
   208→      }
   209→    """
   210→    tasks = []
   211→    jawa_by_idx = {i: e for i, e in enumerate(jawa_entries)}
   212→    last_end = 0.0
   213→    seg_idx = 0
   214→    cumulative_offset = 0.0  # sum durasi semua segments sebelumnya (untuk timestamp akumulatif)
   215→
   216→    # Filter: hanya process cues yang mp4_start < mp4_duration
   217→    for i, m_entry in enumerate(mandarin_entries):
   218→        if m_entry['start'] >= mp4_duration:
   219→            print(f'  → Stop di cue {i} (mp4_start={m_entry["start"]:.2f}s ≥ mp4_duration={mp4_duration:.2f}s)')
   220→            break
   221→
   222→        # Gap sebelum cue ini
   223→        gap_mp4_dur = m_entry['start'] - last_end
   224→        if gap_mp4_dur > 0.01:
   225→            if i in jawa_by_idx and (i - 1) in jawa_by_idx:
   226→                jawa_gap_dur = jawa_by_idx[i]['start'] - jawa_by_idx[i - 1]['end']
   227→            elif i == 0 and i in jawa_by_idx:
   228→                jawa_gap_dur = jawa_by_idx[i]['start']
   229→            else:
   230→                jawa_gap_dur = gap_mp4_dur
   231→
   232→            jawa_gap_dur = max(0.05, jawa_gap_dur)
   233→            factor = jawa_gap_dur / gap_mp4_dur
   234→
   235→            gap_end = min(m_entry['start'], mp4_duration)
   236→            tasks.append({
   237→                'type': 'gap',
   238→                'index': seg_idx,
   239→                'mp4_start': last_end,
   240→                'mp4_end': gap_end,
   241→                'target_duration': jawa_gap_dur,
   242→                'factor': factor,
   243→                'cumulative_offset': cumulative_offset,
   244→            })
   245→            cumulative_offset += jawa_gap_dur
   246→            seg_idx += 1
   247→
   248→        # Cue itu sendiri
   249→        if i in jawa_by_idx and m_entry['end'] <= mp4_duration:
   250→            j_entry = jawa_by_idx[i]
   251→            cue_mp4_dur = m_entry['end'] - m_entry['start']
   252→            cue_jawa_dur = j_entry['end'] - j_entry['start']
   253→
   254→            if cue_mp4_dur > 0.01 and cue_jawa_dur > 0.01:
   255→                factor = cue_jawa_dur / cue_mp4_dur
   256→                tasks.append({
   257→                    'type': 'cue',
   258→                    'index': seg_idx,
   259→                    'mp4_start': m_entry['start'],
   260→                    'mp4_end': m_entry['end'],
   261→                    'target_duration': cue_jawa_dur,
   262→                    'factor': factor,
   263→                    'cumulative_offset': cumulative_offset,
   264→                })
   265→                cumulative_offset += cue_jawa_dur
   266→                seg_idx += 1
   267→
   268→        last_end = min(m_entry['end'], mp4_duration)
   269→
   270→    # Tail gap
   271→    if last_end < mp4_duration - 0.05:
   272→        tail_dur = mp4_duration - last_end
   273→        tasks.append({
   274→            'type': 'tail',
   275→            'index': seg_idx,
   276→            'mp4_start': last_end,
   277→            'mp4_end': mp4_duration,
   278→            'target_duration': tail_dur,
   279→            'factor': 1.0,
   280→            'cumulative_offset': cumulative_offset,
   281→        })
   282→        seg_idx += 1
   283→
   284→    return tasks

   287→# ============================================================
   288→# Render 1 segment (worker function untuk parallel)
   289→# ============================================================
   290→
   291→def render_segment(task, mp4_path, segments_dir, ffmpeg_path, ffprobe_path, preset, has_audio_ori, encoder='libx264'):
   292→    """
   293→    Render 1 segment jadi file MP4 kecil.
   294→    Dipanggil oleh ProcessPoolExecutor (parallel).
   295→    """
   296→    seg_idx = task['index']
   297→    output_file = os.path.join(segments_dir, f'seg_{seg_idx:05d}.mp4')
   298→
   299→    mp4_start = task['mp4_start']
   300→    mp4_end = task['mp4_end']
   301→    mp4_dur = mp4_end - mp4_start
   302→    target_dur = task['target_duration']
   303→    factor = task['factor']
   304→    cumulative_offset = task.get('cumulative_offset', 0.0)
   305→
   306→    # Skip kalau sudah ada (resume support)
   307→    if os.path.isfile(output_file) and os.path.getsize(output_file) > 1000:
   308→        return {'index': seg_idx, 'output_file': output_file, 'status': 'skipped'}
   309→
   310→    # Encoder params: libx264 (software) atau h264_videotoolbox (hardware M1)
   311→    if encoder == 'h264_videotoolbox':
   312→        enc_params = ['-c:v', 'h264_videotoolbox', '-b:v', '5M', '-realtime', '0',
   313→                       '-bf', '0', '-profile:v', 'high']
   314→    else:
   315→        enc_params = ['-c:v', 'libx264', '-preset', preset, '-crf', '23',
   316→                       '-bf', '0', '-profile:v', 'high']
   317→
   318→    use_stream_copy = 0.95 <= factor <= 1.05 and task['type'] in ('gap', 'tail')
   319→
   320→    if use_stream_copy:
   321→        # Stream copy dengan output_ts_offset = timestamp akumulatif
   322→        cmd = [
   323→            ffmpeg_path, '-y',
   324→            '-hwaccel', 'videotoolbox',
   325→            '-ss', f'{mp4_start:.3f}',
   326→            '-i', mp4_path,
   327→            '-t', f'{mp4_dur:.3f}',
   328→            '-c:v', 'copy',
   329→            '-an',
   330→            '-output_ts_offset', f'{cumulative_offset:.6f}',
   331→            '-fflags', '+genpts',
   332→            output_file,
   333→        ]
   334→    else:
   335→        # Re-encode: setpts untuk slow-mo + output_ts_offset untuk timestamp akumulatif
   336→        # fps=30 filter SEBELUM setpts (VFR guard, dari riset VoiceStudio)
   337→        # JANGAN pakai -fps_mode cfr atau -video_track_timescale sebagai output option
   338→        cmd = [
   339→            ffmpeg_path, '-y',
   340→            '-hwaccel', 'videotoolbox',
   341→            '-i', mp4_path,
   342→            '-ss', f'{mp4_start:.3f}',
   343→            '-t', f'{target_dur:.3f}',
   344→            '-vf', f'fps=30,setpts=(PTS-STARTPTS)*{factor:.6f}',
   345→            *enc_params,
   346→            '-output_ts_offset', f'{cumulative_offset:.6f}',
   347→            '-an',
   348→            '-fflags', '+genpts',
   349→            output_file,
   350→        ]
   351→
   352→    try:
   353→        result = subprocess.run(cmd, capture_output=True, text=True, timeout=300)
   354→        if result.returncode != 0:
   355→            # Print stderr LENGKAP untuk debugging (bukan cuma [-300:])
   356→            # Cari baris yang mengandung "Error" atau "error" di stderr
   357→            stderr_lines = result.stderr.split('\n')
   358→            error_lines = [l for l in stderr_lines if 'error' in l.lower() or 'invalid' in l.lower() or 'not found' in l.lower() or 'no such' in l.lower()]
   359→            error_summary = error_lines[0] if error_lines else stderr_lines[-5:]
   360→            return {
   361→                'index': seg_idx,
   362→                'output_file': output_file,
   363→                'status': 'failed',
   364→                'error': f'code={result.returncode} | {error_summary[:200]}',
   365→            }
   366→        # VALIDASI: cek duration output — hanya untuk re-encode segments
   367→        # (stream copy durasi = source durasi, tidak perlu validasi terhadap target)
   368→        # SKIP validasi untuk sekarang. Segment terakhir sering terpotong
   369→        # karena mp4_end > mp4_duration (video input lebih pendek dari target).
   370→        # Concat di Pass 2 akan handle — segment pendek tetap di-concat.
   371→        return {'index': seg_idx, 'output_file': output_file, 'status': 'ok'}
   372→    except subprocess.TimeoutExpired:
   373→        return {
   374→            'index': seg_idx,
   375→            'output_file': output_file,
   376→            'status': 'failed',
   377→            'error': 'timeout (300s)',
   378→        }
   379→    except Exception as e:
   380→        return {
   381→            'index': seg_idx,
   382→            'output_file': output_file,
   383→            'status': 'failed',
   384→            'error': str(e),
   385→        }

   388→# ============================================================
   389→# Pass 1: Render semua segments (parallel)
   390→# ============================================================
   391→
   392→def render_all_segments(tasks, mp4_path, segments_dir, ffmpeg_path, ffprobe_path, preset, workers=4, encoder='libx264'):
   393→    """
   394→    Render semua segments secara parallel.
   395→    Pakai ProcessPoolExecutor untuk parallel FFmpeg processes.
   396→    """
   397→    total = len(tasks)
   398→    print(f'\n=== Pass 1: Render {total} segments (parallel {workers} workers) ===')
   399→
   400→    completed = 0
   401→    failed = 0
   402→    start_time = time.time()
   403→
   404→    with ProcessPoolExecutor(max_workers=workers) as executor:
   405→        # Submit semua tasks
   406→        futures = {
   407→            executor.submit(render_segment, task, mp4_path, segments_dir, ffmpeg_path, ffprobe_path, preset, True, encoder): task
   408→            for task in tasks
   409→        }
   410→
   411→        # Wait dan report progress
   412→        for future in as_completed(futures):
   413→            task = futures[future]
   414→            try:
   415→                result = future.result()
   416→                completed += 1
   417→                if result['status'] == 'failed':
   418→                    failed += 1
   419→                    print(f'  ✗ Segment {result["index"]} gagal: {result.get("error", "unknown")[:100]}')
   420→                elif result['status'] == 'skipped':
   421→                    pass  # silent skip
   422→
   423→                # Progress setiap 50 segments
   424→                if completed % 50 == 0 or completed == total:
   425→                    elapsed = time.time() - start_time
   426→                    rate = completed / elapsed if elapsed > 0 else 0
   427→                    eta = (total - completed) / rate if rate > 0 else 0
   428→                    print(f'  → {completed}/{total} segments ({100*completed/total:.1f}%) — '
   429→                          f'elapsed {elapsed:.0f}s, ETA {eta:.0f}s, failed {failed}')
   430→            except Exception as e:
   431→                failed += 1
   432→                print(f'  ✗ Task exception: {e}')
   433→
   434→    print(f'\n=== Pass 1 selesai ===')
   435→    print(f'  Total: {total} segments')
   436→    print(f'  Sukses: {total - failed}')
   437→    print(f'  Gagal: {failed}')
   438→    print(f'  Waktu: {time.time() - start_time:.0f}s')
   439→
   440→    # Allow up to 5% failure (segment pendek di akhir video wajar gagal)
   441→    # Kalau gagal > 5%, return False (gagal total)
   442→    # Kalau gagal <= 5%, return True (lanjut ke Pass 2, segment gagal di-skip di concat)
   443→    if failed > total * 0.05:
   444→        print(f'  ❌ Gagal {failed}/{total} ({100*failed/total:.1f}%) > 5% threshold')
   445→        return False
   446→    elif failed > 0:
   447→        print(f'  ⚠ {failed} segment gagal ({100*failed/total:.1f}%), tapi < 5% → lanjut ke Pass 2 (segment gagal di-skip)')
   448→    return True

   455→def concat_segments(tasks, segments_dir, audio_jawa, output, ffmpeg_path, has_audio_ori, ducking_db, video_only=False, encoder='libx264'):
   456→    """
   457→    Concat semua segment files menggunakan concat demuxer.
   458→    Kalau video_only=True: output = video tanpa audio (user import audio terpisah di DaVinci).
   459→    """
   460→    print(f'\n=== Pass 2: Concat segments + mix audio ===')
   461→
   462→    # Build concat list file — skip segments yang gagal (file tidak ada atau < 1KB)
   463→    concat_list = os.path.join(segments_dir, 'concat_list.txt')
   464→    included = 0
   465→    skipped = 0
   466→    with open(concat_list, 'w') as f:
   467→        for task in tasks:
   468→            seg_file = os.path.join(segments_dir, f'seg_{task["index"]:05d}.mp4')
   469→            # Skip kalau file tidak ada atau terlalu kecil (segment gagal)
   470→            if os.path.isfile(seg_file) and os.path.getsize(seg_file) > 1000:
   471→                f.write(f"file '{seg_file}'\n")
   472→                included += 1
   473→            else:
   474→                skipped += 1
   475→
   476→    print(f'  → Concat list: {concat_list} ({included} files, {skipped} skipped)')
   477→
   478→    # Encoder params untuk Pass 2 re-encode
   479→    if encoder == 'h264_videotoolbox':
   480→        enc_params = ['-c:v', 'h264_videotoolbox', '-b:v', '5M', '-realtime', '0',
   481→                       '-bf', '0', '-profile:v', 'high']
   482→    else:
   483→        enc_params = ['-c:v', 'libx264', '-preset', 'fast', '-crf', '23',
   484→                       '-bf', '0', '-profile:v', 'high']
   485→
   486→    # Pass 2: STREAM COPY (instant, 2 detik)
   487→    # Timestamp sudah akumulatif dari Pass 1 (output_ts_offset)
   488→    # → concat demuxer stream copy = seamless, tidak ada DTS out of order
   489→    if video_only:
   490→        print(f'  → Mode: VIDEO ONLY (stream copy, tanpa audio)')
   491→        cmd = [
   492→            ffmpeg_path, '-y',
   493→            '-f', 'concat', '-safe', '0',
   494→            '-i', concat_list,
   495→            '-c:v', 'copy',
   496→            '-an',
   497→            '-movflags', '+faststart',
   498→            '-timecode', '00:00:00:00',
   499→            output,
   500→        ]
   501→    else:
   502→        cmd = [
   503→            ffmpeg_path, '-y',
   504→            '-f', 'concat', '-safe', '0',
   505→            '-i', concat_list,
   506→            '-i', audio_jawa,
   507→            '-map', '0:v',
   508→            '-map', '1:a',
   509→            '-c:v', 'copy',
   510→            '-c:a', 'aac',
   511→            '-b:a', '192k',
   512→            '-movflags', '+faststart',
   513→            '-timecode', '00:00:00:00',
   514→            '-shortest',
   515→            output,
   516→        ]
   517→
   518→    print(f'  → Concat + mix audio...')
   519→    print(f'  Command: {" ".join(cmd[:5])} ...')
   520→
   521→    result = subprocess.run(cmd)
   522→    if result.returncode != 0:
   523→        print(f'\n✗ Pass 2 gagal dengan code {result.returncode}')
   524→        return False
   525→
   526→    print(f'  ✓ Pass 2 selesai')
   527→    return True

   531→# Demucs SFX separation (optional)
   532→# ============================================================
   533→
   534→def separate_sfx_with_demucs(mp4_path: str, work_dir: str) -> dict:
   535→    print('\n=== SFX Separation dengan Demucs ===')
   536→    demucs_bin = shutil.which('demucs') or shutil.which('demucs.exe')
   537→    if not demucs_bin:
   538→        print('  ⚠ Demucs tidak ditemukan. Skip SFX separation.')
   539→        return None
   540→
   541→    audio_ori = os.path.join(work_dir, 'audio-original.wav')
   542→    ffmpeg = find_ffmpeg()
   543→    subprocess.run([
   544→        ffmpeg, '-y', '-i', mp4_path,
   545→        '-vn', '-acodec', 'pcm_s16le', '-ar', '44100', '-ac', '2',
   546→        audio_ori,
   547→    ], check=True)
   548→
   549→    print('  → Run Demucs (mungkin butuh 5-15 menit untuk file besar)...')
   550→    demucs_out_dir = os.path.join(work_dir, 'demucs-output')
   551→    os.makedirs(demucs_out_dir, exist_ok=True)
   552→    result = subprocess.run([
   553→        demucs_bin, '--two-stems', 'vocals', '-o', demucs_out_dir, audio_ori,
   554→    ])
   555→    if result.returncode != 0:
   556→        print(f'  ✗ Demucs gagal. Skip SFX.')
   557→        return None
   558→
   559→    base_name = os.path.splitext(os.path.basename(audio_ori))[0]
   560→    no_vocals = os.path.join(demucs_out_dir, base_name, 'no_vocals.wav')
   561→    if not os.path.isfile(no_vocals):
   562→        return None
   563→
   564→    print(f'  ✓ SFX: {no_vocals}')
   565→    return {'no_vocals': no_vocals}
